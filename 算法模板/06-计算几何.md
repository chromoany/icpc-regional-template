# 06 · 计算几何

> 约定：点集用全局数组 `px[], py[]` 存储（下标），函数传下标；叉积用 ll（坐标 $\le 10^9$ 时安全，更大改用 __int128 或 long double）。

## 1. 基础运算：叉积 / 点积 / 线段相交 / 点在多边形内

适用：判断点的方位、两线段是否相交、点是否在多边形内等基础判定。

复杂度：$O(1)$ / $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
ll px[MAXN], py[MAXN];
int n;

// 符号函数：x>0 返回 1，x<0 返回 -1，x==0 返回 0
int sgn(ll x)
{
    if (x > 0) return 1;
    if (x < 0) return -1;
    return 0;
}

// 向量叉积 (x1,y1)×(x2,y2)；>0 表示后一向量在前一向量的逆时针方向（左），=0 共线
ll cross(ll x1, ll y1, ll x2, ll y2)
{
    return x1 * y2 - y1 * x2;
}

// 向量点积 (x1,y1)·(x2,y2)；>0 夹角为锐角，=0 垂直，<0 为钝角
ll dot(ll x1, ll y1, ll x2, ll y2)
{
    return x1 * x2 + y1 * y2;
}

// 判断点 (x,y) 是否在线段 P1(x1,y1)-P2(x2,y2) 上（含端点）
bool on_seg(ll x1, ll y1, ll x2, ll y2, ll x, ll y)
{
    return min(x1, x2) <= x && x <= max(x1, x2) && min(y1, y2) <= y && y <= max(y1, y2)
        && (x2 - x1) * (y - y1) == (y2 - y1) * (x - x1);
}

// 判断两线段 P1P2 与 P3P4 是否相交（含端点恰落在另一线段上的情况）
bool seg_intersect(ll x1, ll y1, ll x2, ll y2,
                   ll x3, ll y3, ll x4, ll y4)
{
    ll d1 = cross(x2 - x1, y2 - y1, x3 - x1, y3 - y1);
    ll d2 = cross(x2 - x1, y2 - y1, x4 - x1, y4 - y1);
    ll d3 = cross(x4 - x3, y4 - y3, x1 - x3, y1 - y3);
    ll d4 = cross(x4 - x3, y4 - y3, x2 - x3, y2 - y3);
    if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) && ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0))) return true;
    if (d1 == 0 && on_seg(x1, y1, x2, y2, x3, y3)) return true;
    if (d2 == 0 && on_seg(x1, y1, x2, y2, x4, y4)) return true;
    if (d3 == 0 && on_seg(x3, y3, x4, y4, x1, y1)) return true;
    if (d4 == 0 && on_seg(x3, y3, x4, y4, x2, y2)) return true;
    return false;
}

// 射线法判断点 (x,y) 是否在全局多边形 (px[1..n], py[1..n]) 内部；只判内部，不含边界
bool point_in_poly(ll x, ll y)
{
    bool inside = false;
    for (int i = 1, j = n; i <= n; j = i++)
    {
        if ((py[i] > y) != (py[j] > y))
        {
            long double xc = (long double)(px[j] - px[i]) * (y - py[i]) / (py[j] - py[i]) + px[i];
            if (x < xc) inside = !inside;
        }
    }
    return inside;
}
```

使用说明：
- `cross(a, b, c)` 用向量差：`cross(px[b]-px[a], py[b]-py[a], px[c]-px[a], py[c]-py[a])`，$>0$ 表示 c 在 ab 左侧。
- `point_in_poly` 为射线法（含边界需自行加判定：先用 on_seg 判断点是否在边上）。
- 坐标乘积可能溢出 ll 时，把叉积函数内类型换成 `__int128`。

变体：
- 三点共线：`cross == 0`。
- 点在线段上：`cross == 0` 且坐标在范围内（on_seg）。
- 直线交点 / 点到直线距离：用叉积与面积公式。

## 2. 凸包：Andrew 算法

适用：平面点集求凸包（按 x,y 排序，单调栈两次扫描）。

复杂度：$O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
ll px[MAXN], py[MAXN];
int ord[MAXN], stk[MAXN];
int n, top;

// 叉积：向量 (o→a) × (o→b)；>0 表示 b 在 oa 的逆时针方向（左），=0 三点共线
ll cross3(int o, int a, int b)
{
    return (px[a] - px[o]) * (py[b] - py[o]) - (py[a] - py[o]) * (px[b] - px[o]);
}

// 点排序比较函数：按 x 升序，x 相同按 y 升序
bool cmp_point(int a, int b)
{
    if (px[a] != px[b]) return px[a] < px[b];
    return py[a] < py[b];
}

// Andrew 求凸包：结果顶点下标存 stk[1..top]（逆时针，首尾不重复）；top<3 表示共线或点数不足
void andrew()
{
    for (int i = 1; i <= n; i++) ord[i] = i;
    sort(ord + 1, ord + n + 1, cmp_point);
    top = 0;
    for (int i = 1; i <= n; i++)
    {
        while (top >= 2 && cross3(stk[top - 1], stk[top], ord[i]) <= 0)
        {
            top--;
        }
        stk[++top] = ord[i];
    }
    int lower = top;
    for (int i = n - 1; i >= 1; i--)
    {
        while (top > lower && cross3(stk[top - 1], stk[top], ord[i]) <= 0)
        {
            top--;
        }
        stk[++top] = ord[i];
    }
    top--;
}

// 凸包曼哈顿周长：沿 stk[1..top] 相邻顶点累加 |dx|+|dy|
ll convex_perimeter()
{
    ll per = 0;
    for (int i = 1; i <= top; i++) per += abs(px[stk[i]] - px[stk[i % top + 1]]) + abs(py[stk[i]] - py[stk[i % top + 1]]);
    return per;
}
```

使用说明：
- 调用 `andrew()` 后 `stk[1..top]` 为凸包顶点（逆时针），`top < 3` 时共线 / 点数不足。
- 共线点处理：`<= 0` 弹栈会去掉边上点，保留端点；要保留边上点改用 `< 0`。
- 周长、面积、直径（旋转卡壳）基于 stk 序列。

变体：
- 凸包旋转卡壳求直径：对凸包双指针扫，$O(n)$。
- 凸包内最大面积三角形：旋转卡壳三重指针。

## 3. 多边形面积 / 最近点对

### 3a. 多边形面积（鞋带公式）

适用：任意简单多边形面积（顶点按顺序，逆/顺时针均可）。

复杂度：$O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
ll px[MAXN], py[MAXN];
int n;

// 鞋带公式求多边形面积（顶点按 px[1..n],py[1..n] 顺序）：返回的是 2 倍面积，真实面积需除以 2
ll polygon_area()
{
    ll area = 0;
    for (int i = 1; i <= n; i++)
    {
        int j = (i == n ? 1 : i + 1);
        area += px[i] * py[j] - py[i] * px[j];
    }
    if (area < 0) area = -area;
    return area;
}
```

使用说明：
- 返回面积（实际为面积的 2 倍，若需真实面积除以 2 用 long double）。
- 顶点需按顺序排列（存储顺序即可）。

变体：
- 凸多边形重心：加权三角形质心。

### 3b. 最近点对（分治）

适用：平面 n 个点求最近距离（返回平方距离，开方得距离）。

复杂度：$O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
ll px[MAXN], py[MAXN];
int ord[MAXN], tmpn[MAXN];
int n;

// 平方
ll sq(ll x)
{
    return x * x;
}

// 两点欧氏距离的平方（最终距离 = sqrt(dist2(a,b))）
ll dist2(int a, int b)
{
    return sq(px[a] - px[b]) + sq(py[a] - py[b]);
}

// 点排序比较函数：按 x 升序，x 相同按 y 升序（最近点对分治前先排好序）
bool cmp_point(int a, int b)
{
    if (px[a] != px[b]) return px[a] < px[b];
    return py[a] < py[b];
}

// 按 y 升序（分治中条带内点排序用）
bool cmpy(int a, int b)
{
    return py[a] < py[b];
}

// 分治求最近点对：要求 ord[l..r] 已按 x 排序，返回最近平方距离（无点对返回 INF）
ll closest(int l, int r)
{
    if (l >= r) return INF;
    if (r - l == 1) return dist2(ord[l], ord[r]);
    int mid = (l + r) >> 1;
    ll midx = px[ord[mid]];
    ll d = min(closest(l, mid), closest(mid + 1, r));
    int k = 0;
    for (int i = l; i <= r; i++) if (sq(px[ord[i]] - midx) < d) tmpn[k++] = ord[i];
    sort(tmpn, tmpn + k, cmpy);
    for (int i = 0; i < k; i++) for (int j = i + 1; j < k && sq(py[tmpn[j]] - py[tmpn[i]]) < d; j++) d = min(d, dist2(tmpn[i], tmpn[j]));
    return d;
}
```

使用说明：
- 先对 `ord[1..n]` 按 x 排序（`cmp_point`），再调用 `closest(1, n)`。
- 返回最近点对的平方距离，`sqrt` 后为实际距离。
- 坐标用 ll，平方后需 $\le 9\times10^{18}$（坐标 $\le 10^9$ 内安全）。

变体：
- 最近点对 + 平面扫描（set 维护）也 $O(n\log n)$。
- 最近点对的"带权"版（曼哈顿 / 切比雪夫距离）需变换坐标。

## 4. 是否存在直线穿过所有线段（刺穿直线）

适用：给定 n 条线段，判断是否存在一条直线与每条线段都相交（擦到线段端点也算穿过）。等价表述（POJ 3304 原题）：是否存在一条直线，使所有线段在它上面的投影有公共点（公共投影点处作垂线即穿过所有线段的直线）。

核心推导：若可行直线存在，则必存在一条同样穿过所有线段、且经过两个线段端点（两端点可属同一条线段）的直线——把解直线沿其法向平移，直到某条线段的端点贴上直线（平移过程仍穿过全部线段），再绕该端点旋转，直到再次贴住另一端点为止。故只需枚举 2n 个端点两两确定的直线逐一验证。单条直线 ab 的判定：线段 i 两端点 u、v 对 ab 的叉积同正或同负（分别大于 EPS / 小于 -EPS）说明线段整体在直线一侧、不相交；否则相交（含端点恰在直线上）。

复杂度：$O(n^3)$（$\binom{2n}{2}$ 条候选直线 $\times$ 每条 $O(n)$ 判定），空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 100005;
const double EPS = 1e-8;

int n;
double px[MAXN], py[MAXN];

double cross(double x1, double y1, double x2, double y2)
{
    return x1 * y2 - y1 * x2;
}

bool check_line(int a, int b)
{
    double ux = px[b] - px[a], uy = py[b] - py[a];
    for (int i = 1; i <= n; i++)
    {
        int u = 2 * i - 1, v = 2 * i;
        double d1 = cross(ux, uy, px[u] - px[a], py[u] - py[a]);
        double d2 = cross(ux, uy, px[v] - px[a], py[v] - py[a]);
        if (d1 > EPS && d2 > EPS) return false;
        if (d1 < -EPS && d2 < -EPS) return false;
    }
    return true;
}
```

使用说明：
- 所有线段端点依次读入全局 `px[1..2n] / py[1..2n]`，线段 i 的两个端点是下标 `2i-1` 与 `2i`，与本文件开头的点集约定一致。
- main 内流程：读 T 组数据，每组读 n 与端点；`n <= 2` 直接输出 Yes（任取两条线段上各一点连线即可）；若 2n 个端点全部重合为一点（全部退化成同一单点）也直接 Yes；否则双重循环枚举端点对 `a < b`（重合端点对跳过），`check_line(a, b)` 为真则 Yes，全部失败则 No。
- 坐标为实数时用 double + EPS；若坐标全为整数且绝对值 $\le 10^9$，可整体换 long long 叉积并去掉 EPS。
- 竖直线无需特判：叉积判定不依赖斜率。

变体：
- 若题意要求严格穿过线段内部（碰到端点不算穿过），端点对枚举思路失效，不能直接套本模板。
- 线段换成折线 / 管道（如 HDU 2150：每根管道为一段折线）：端点池换成所有折线的顶点，判定改为"直线与折线中任一折线段相交"。
- 需要输出一条可行直线：记录成功的那对端点 a、b，输出直线 ab 即可。

## 5. 位置关系与距离：点线 / 点圆 / 线圆 / 圆圆

适用：判断点与直线（左侧/右侧/在线上）、点与圆（圆内/圆上/圆外）、直线与圆（相交/相切/相离）、圆与圆（外离/外切/相交/内切/内含/重合）的位置关系，并求两点、点到直线、点到线段的距离。

复杂度：$O(1)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

// 符号函数：x>0 返回 1，x<0 返回 -1，x==0 返回 0
int sgn(ll x)
{
    if (x > 0) return 1;
    if (x < 0) return -1;
    return 0;
}

// 向量叉积 (x1,y1)×(x2,y2)；>0 逆时针（左），=0 共线
ll cross(ll x1, ll y1, ll x2, ll y2)
{
    return x1 * y2 - y1 * x2;
}

// 向量点积 (x1,y1)·(x2,y2)
ll dot(ll x1, ll y1, ll x2, ll y2)
{
    return x1 * x2 + y1 * y2;
}

// 两点欧氏距离
long double dist_pp(ll x1, ll y1, ll x2, ll y2)
{
    long double dx = x1 - x2, dy = y1 - y2;
    return sqrtl(dx * dx + dy * dy);
}

// 点 (x,y) 与直线 AB（A、B 确定无限直线）的位置关系：返回叉积符号
// 1 = 在 AB 左侧，-1 = 右侧，0 = 在直线上
int rel_point_line(ll x, ll y, ll x1, ll y1, ll x2, ll y2)
{
    return sgn(cross(x2 - x1, y2 - y1, x - x1, y - y1));
}

// 点 (x,y) 到直线 AB 的距离
long double dist_pl(ll x, ll y, ll x1, ll y1, ll x2, ll y2)
{
    long double vx = x2 - x1, vy = y2 - y1;
    long double wx = x - x1, wy = y - y1;
    return fabsl(vx * wy - vy * wx) / sqrtl(vx * vx + vy * vy);
}

// 点 (x,y) 到线段 AB 的距离（垂足在线段外时取最近端点）
long double dist_ps(ll x, ll y, ll x1, ll y1, ll x2, ll y2)
{
    ll vx = x2 - x1, vy = y2 - y1;
    if (dot(vx, vy, x - x1, y - y1) <= 0) return dist_pp(x, y, x1, y1);
    if (dot(vx, vy, x - x2, y - y2) >= 0) return dist_pp(x, y, x2, y2);
    return dist_pl(x, y, x1, y1, x2, y2);
}

// 点 (x,y) 与圆 C(cx,cy,r) 的位置关系：-1 圆内，0 圆上，1 圆外
int rel_point_circle(ll x, ll y, ll cx, ll cy, ll r)
{
    __int128 dx = (__int128)x - cx, dy = (__int128)y - cy;
    __int128 d2 = dx * dx + dy * dy;
    __int128 r2 = (__int128)r * r;
    if (d2 < r2) return -1;
    if (d2 == r2) return 0;
    return 1;
}

// 直线 AB 与圆 C(cx,cy,r) 的位置关系（精确整数判定，无 EPS）
// 圆心到直线距离 d = |cross(AB,AC)|/|AB|，比较 d 与 r 等价于比较 cross² 与 r²·|AB|²
// -1 相交（两个交点），0 相切，1 相离
int rel_line_circle(ll x1, ll y1, ll x2, ll y2, ll cx, ll cy, ll r)
{
    __int128 cr = ((__int128)x2 - x1) * ((__int128)cy - y1) - ((__int128)y2 - y1) * ((__int128)cx - x1);
    __int128 ab2 = ((__int128)x2 - x1) * ((__int128)x2 - x1) + ((__int128)y2 - y1) * ((__int128)y2 - y1);
    __int128 delta = cr * cr - (__int128)r * r * ab2;
    if (delta < 0) return -1;
    if (delta > 0) return 1;
    return 0;
}

// 圆 C1(cx1,cy1,r1) 与圆 C2(cx2,cy2,r2) 的位置关系（d 为圆心距）
// 0 外离（d>r1+r2） 1 外切（d==r1+r2） 2 相交（|r1-r2|<d<r1+r2）
// 3 内切（d==|r1-r2|） 4 内含（d<|r1-r2|） 5 重合（d==0 且 r1==r2）
int rel_circle_circle(ll cx1, ll cy1, ll r1, ll cx2, ll cy2, ll r2)
{
    __int128 dx = (__int128)cx1 - cx2, dy = (__int128)cy1 - cy2;
    __int128 d2 = dx * dx + dy * dy;
    __int128 sum = (__int128)r1 + r2;
    __int128 sub = r1 > r2 ? (__int128)r1 - r2 : (__int128)r2 - r1;
    if (d2 == 0 && r1 == r2) return 5;
    if (d2 > sum * sum) return 0;
    if (d2 == sum * sum) return 1;
    if (d2 == sub * sub) return 3;
    if (d2 < sub * sub) return 4;
    return 2;
}
```

使用说明：
- 本节"关系"判定全部基于整数精确比较（中间乘法用 `__int128`），没有 EPS、不出现边界误判；点、直线、圆的坐标与半径需以整数读入。
- 返回值汇总：点线 1 左 / -1 右 / 0 在直线上（把 AB 看作有向直线，返回叉积符号）；点圆 -1 圆内 / 0 圆上 / 1 圆外；线圆 -1 相交（两交点）/ 0 相切 / 1 相离；圆圆 0 外离 / 1 外切 / 2 相交 / 3 内切 / 4 内含 / 5 重合。
- `rel_line_circle` 原理：距离 $d=|\text{cross}(AB,AC)|/|AB|$，比较 $d$ 与 $r$ $\iff$ 比较 $\text{cross}^2$ 与 $r^2\cdot|AB|^2$（$|AB|^2>0$，两边同乘不等号不变），从而完全避开浮点。
- 距离函数返回 long double（先转 long double 再平方，避免溢出），输出用 `cout << fixed << setprecision(...)` 或 `printf("%.6Lf", ...)`。
- `rel_point_line` / `dist_pl` 要求 A、B 不重合（直线退化无意义）；`dist_ps` 无此限制，自动退化为点到端点距离。
- 半径可为 0（点圆退化）：两圆都退化为点时，`rel_circle_circle` 仍正确给出外离 / 重合。
- 附带可推距离：点与圆最短距离 $=\max(0.0L,\ \text{dist\_pp}-r)$；两圆最短距离 $=\max(0.0L,\ \text{圆心距}-r_1-r_2)$。

变体：
- 坐标为实数（读 double）时：签名里的 ll 全换 double，比较处加 EPS，例如线圆：`d = fabsl(vx*wy-vy*wx)/sqrtl(vx*vx+vy*vy)`，`d < r - EPS` 相交、`fabs(d - r) <= EPS` 相切、否则相离；圆圆用 $d$ 与 $r_1+r_2$、$|r_1-r_2|$ 加减 EPS 比较。
- 只要"直线与圆是否相交（含相切）"：判 `rel_line_circle(...) != 1`；只要"两圆是否相交（含相切）"：判 `rel_circle_circle(...) == 2`。
- 相切 / 重合 / 内含等边界情况本模板整数下精确区分；若改成 double 版这些点必须靠 EPS 兜底。
- 需要交点坐标时：线圆交点为圆心向直线所作垂线的垂足沿直线方向平移 $\pm\sqrt{r^2-d^2}$；圆圆交点在连心线上取距 $C_1$ 为 $a=\dfrac{d^2+r_1^2-r_2^2}{2d}$ 的点再沿垂直方向平移 $\pm\sqrt{r_1^2-a^2}$。
