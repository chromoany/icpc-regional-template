# 03 · 图论

## 1. Dijkstra 堆优化（单源最短路）

适用：正权图单源最短路。稠密图可换朴素版。

复杂度：$O((n+m)\log n)$，空间 $O(n+m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
vector<pair<int, int>> g[MAXN];
ll dis[MAXN];
int vis[MAXN];
int n, m;

void add_edge(int u, int v, int ww)
{
    g[u].push_back({v, ww});
}

void dijkstra(int s)
{
    priority_queue<pair<ll, int>, vector<pair<ll, int>>, greater<pair<ll, int>>> pq;
    for (int i = 1; i <= n; i++)
    {
        dis[i] = INF;
        vis[i] = 0;
    }
    dis[s] = 0;
    pq.push({0, s});
    while (!pq.empty())
    {
        int u = pq.top().second;
        pq.pop();
        if (vis[u]) continue;
        vis[u] = 1;
        for (auto ed : g[u])
        {
            int v = ed.first;
            int w = ed.second;
            if (dis[v] > dis[u] + w)
            {
                dis[v] = dis[u] + w;
                pq.push({dis[v], v});
            }
        }
    }
}
```

使用说明：
- 加边：`add_edge(u,v,w)` 加一条有向边；无向图再加 `add_edge(v,u,w)`。
- 结果存 `dis[]`，不可达为 INF。
- 求路径：加全局 `pre[v]` 记录松弛来源点，从 t 反向回溯。

变体：
- 求第 k 短路：A*（估价函数用反图最短路）或可持久化左偏树。
- 分层图最短路：拆点 (u, 层)，层内同权、跨层走特殊边，仍 Dijkstra。
- 稠密图 $n\le 2000$：朴素 $O(n^2)$ 更快。

## 2. SPFA（负权 / 判负环）

适用：存在负权边时 Bellman-Ford 的队列优化，或需要判负环。

复杂度：平均 $O(m)$，最坏 $O(nm)$。判负环用入队次数。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int INF = 0x3f3f3f3f;
vector<pair<int, int>> g[MAXN];
int dis[MAXN];
int cnt[MAXN];
int inq[MAXN];
int n, m;

void add_edge(int u, int v, int ww)
{
    g[u].push_back({v, ww});
}

bool spfa(int s)
{
    queue<int> q;
    for (int i = 1; i <= n; i++)
    {
        dis[i] = INF;
        cnt[i] = 0;
        inq[i] = 0;
    }
    dis[s] = 0;
    q.push(s);
    inq[s] = 1;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        inq[u] = 0;
        for (auto ed : g[u])
        {
            int v = ed.first;
            int w = ed.second;
            if (dis[v] > dis[u] + w)
            {
                dis[v] = dis[u] + w;
                if (!inq[v])
                {
                    q.push(v);
                    inq[v] = 1;
                    if (++cnt[v] > n) return true;
                }
            }
        }
    }
    return false;
}
```

使用说明：
- 返回 true 表示存在从 s 可达的负环。
- 求差分约束：`dis[v] <= dis[u] + w` 对应约束 `x_v - x_u <= w`，加边 u→v 权 w。

变体：
- 判全图负环：先给所有点入队（或建虚拟源点连 0 权边），再跑。
- 负环检测可用 `dfs` 版更早返回，但可能爆栈。

## 3. Floyd（全源最短路）

适用：$n\le 500$ 的全源最短路、传递闭包、最小环。

复杂度：$O(n^3)$，空间 $O(n^2)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 505;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
ll dis[MAXN][MAXN];
int n, m;

void floyd()
{
    for (int k = 1; k <= n; k++) for (int i = 1; i <= n; i++) for (int j = 1; j <= n; j++) if (dis[i][j] > dis[i][k] + dis[k][j]) dis[i][j] = dis[i][k] + dis[k][j];
}
```

使用说明：
- 初始化 `dis[i][j] = INF`（$i\ne j$）、`dis[i][i] = 0`，再读边取 min。
- 传递闭包：权换成布尔可达，`dis[i][j] |= dis[i][k] && dis[k][j]`（可用 bitset 优化到 $O(n^3/64)$）。

变体：
- 最小环：在 Floyd 第 k 层更新前，枚举 i,j 用 `dis[i][j] + w[i][k] + w[k][j]` 更新答案。
- 保存路径：加 `mid[i][j]` 记录中转点递归输出。

## 4. Kruskal 最小生成树

适用：无向带权图求 MST / 最小瓶颈生成树。

复杂度：$O(m\log m)$，空间 $O(n+m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int MAXM = 200005;
int fa[MAXN];
int eu[MAXM], ev[MAXM], ew[MAXM], eid[MAXM];
int n, m;

int find(int x)
{
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

bool cmp_edge(int x, int y)
{
    return ew[x] < ew[y];
}

ll kruskal()
{
    for (int i = 1; i <= n; i++) fa[i] = i;
    for (int i = 1; i <= m; i++) eid[i] = i;
    sort(eid + 1, eid + m + 1, cmp_edge);
    ll ans = 0;
    int cnt = 0;
    for (int i = 1; i <= m; i++)
    {
        int e = eid[i];
        int u = find(eu[e]);
        int v = find(ev[e]);
        if (u != v)
        {
            fa[u] = v;
            ans += ew[e];
            cnt++;
        }
    }
    if (cnt != n - 1) return -1;
    return ans;
}
```

使用说明：
- 边数组 `eu/ev/ew` 存第 i 条边，返回 -1 表示图不连通。
- 需要存 MST 边集时，合并成功时把 `e` 记录到 `in_tree[]`。

变体：
- 次小生成树：Kruskal 后对每条非树边求路径上最大边（树剖 / 倍增），`ans = MST + w - 路径最大边` 取 min。
- 最小瓶颈路：MST 上任意两点路径的最大边即最小瓶颈。
- 优先用 Kruskal 重构树：克鲁斯卡尔树，支持离线"边权 $\le x$ 的连通性/最大值"。

## 5. LCA 倍增

适用：静态树求 LCA、两点距离、路径第 k 个点。可与差分等组合。

复杂度：预处理 $O(n\log n)$，查询 $O(\log n)$，空间 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int LOG = 18;
vector<int> g[MAXN];
int dep[MAXN];
int fa[MAXN][LOG];
int n, q;

void add_edge(int u, int v)
{
    g[u].push_back(v);
}

void dfs(int u, int f)
{
    fa[u][0] = f;
    dep[u] = dep[f] + 1;
    for (int j = 1; j < LOG; j++) fa[u][j] = fa[fa[u][j - 1]][j - 1];
    for (int v : g[u])
    {
        if (v != f) dfs(v, u);
    }
}

int lca(int u, int v)
{
    if (dep[u] < dep[v]) swap(u, v);
    int d = dep[u] - dep[v];
    for (int j = 0; j < LOG; j++) if (d & (1 << j)) u = fa[u][j];
    if (u == v) return u;
    for (int j = LOG - 1; j >= 0; j--)
    {
        if (fa[u][j] != fa[v][j])
        {
            u = fa[u][j];
            v = fa[v][j];
        }
    }
    return fa[u][0];
}

int dist(int u, int v)
{
    int w = lca(u, v);
    return dep[u] + dep[v] - 2 * dep[w];
}
```

使用说明：
- 调用 `dfs(1, 0)`（根的父亲为 0，dep[0]=0）。
- 无向边双向 `add_edge`。
- 路径点权差分：`val[u]++, val[v]++, val[lca]--, val[fa[lca][0]]--`，二次 dfs 收集。

变体：
- 在线 LCA 也可用欧拉序 + ST 表，查询 $O(1)$。
- 动态树（边权修改 / 换根）用树链剖分或 Link-Cut Tree。
- 两点间第 k 个点：跳到 `dep[u]-k` 深度。

## 6. 树链剖分 + 线段树（路径加 / 路径和 / 子树操作）

适用：树上的路径修改 + 路径查询、子树修改 + 子树查询（如路径加、路径和、子树和、子树最值）。

复杂度：预处理 $O(n\log n)$，单次操作 $O(\log^2 n)$，空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
vector<int> g[MAXN];
int fa[MAXN], dep[MAXN], sz[MAXN], son[MAXN];
int top[MAXN], dfn[MAXN], rnk[MAXN], timer;
ll sum[MAXN << 2];
ll lazy[MAXN << 2];
int n, m;

void add_edge(int u, int v)
{
    g[u].push_back(v);
}

void dfs1(int u, int f)
{
    fa[u] = f;
    dep[u] = dep[f] + 1;
    sz[u] = 1;
    son[u] = 0;
    for (int v : g[u])
    {
        if (v == f) continue;
        dfs1(v, u);
        sz[u] += sz[v];
        if (sz[v] > sz[son[u]]) son[u] = v;
    }
}

void dfs2(int u, int tp)
{
    top[u] = tp;
    dfn[u] = ++timer;
    rnk[timer] = u;
    if (son[u]) dfs2(son[u], tp);
    for (int v : g[u])
    {
        if (v != fa[u] && v != son[u]) dfs2(v, v);
    }
}

void push_up(int rt)
{
    sum[rt] = sum[rt << 1] + sum[rt << 1 | 1];
}

void push_down(int rt, int l, int r)
{
    if (lazy[rt])
    {
        int mid = (l + r) >> 1;
        sum[rt << 1] += lazy[rt] * (mid - l + 1);
        sum[rt << 1 | 1] += lazy[rt] * (r - mid);
        lazy[rt << 1] += lazy[rt];
        lazy[rt << 1 | 1] += lazy[rt];
        lazy[rt] = 0;
    }
}

void build(int rt, int l, int r, ll *a)
{
    lazy[rt] = 0;
    if (l == r)
    {
        sum[rt] = a[rnk[l]];
        return;
    }
    int mid = (l + r) >> 1;
    build(rt << 1, l, mid, a);
    build(rt << 1 | 1, mid + 1, r, a);
    push_up(rt);
}

void update(int rt, int l, int r, int ql, int qr, ll v)
{
    if (ql <= l && r <= qr)
    {
        sum[rt] += v * (r - l + 1);
        lazy[rt] += v;
        return;
    }
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(rt << 1, l, mid, ql, qr, v);
    if (qr > mid) update(rt << 1 | 1, mid + 1, r, ql, qr, v);
    push_up(rt);
}

ll query(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return sum[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query(rt << 1, l, mid, ql, qr);
    if (qr > mid) res += query(rt << 1 | 1, mid + 1, r, ql, qr);
    return res;
}

void path_update(int u, int v, ll w)
{
    while (top[u] != top[v])
    {
        if (dep[top[u]] < dep[top[v]]) swap(u, v);
        update(1, 1, n, dfn[top[u]], dfn[u], w);
        u = fa[top[u]];
    }
    if (dep[u] > dep[v]) swap(u, v);
    update(1, 1, n, dfn[u], dfn[v], w);
}

ll path_query(int u, int v)
{
    ll res = 0;
    while (top[u] != top[v])
    {
        if (dep[top[u]] < dep[top[v]]) swap(u, v);
        res += query(1, 1, n, dfn[top[u]], dfn[u]);
        u = fa[top[u]];
    }
    if (dep[u] > dep[v]) swap(u, v);
    res += query(1, 1, n, dfn[u], dfn[v]);
    return res;
}

void subtree_update(int u, ll w)
{
    update(1, 1, n, dfn[u], dfn[u] + sz[u] - 1, w);
}

ll subtree_query(int u)
{
    return query(1, 1, n, dfn[u], dfn[u] + sz[u] - 1);
}
```

使用说明：
- 预处理：`dfs1(1, 0); dfs2(1, 1); build(1, 1, n, val);`，val[i] 为点权数组。
- `path_update/path_query` 操作点权路径；`subtree_*` 操作子树（dfs 序连续）。
- 若为边权：把边权下沉到儿子点，路径查询时排除 LCA（`dfn[son[lca]]` 起查）。

变体：
- 换根操作：换根后子树区间需分类讨论（根在子树外/内）。
- 可配合可持久化 / 动态开点线段树做历史版本。
- 路径最大值 / 最小值：线段树 `push_up` 改取 max/min。

## 7. Tarjan 强连通分量（缩点）

适用：有向图求 SCC，缩点后为 DAG（可用于拓扑、最短路、可达性）。

复杂度：$O(n+m)$，空间 $O(n+m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
vector<int> g[MAXN];
int dfn[MAXN], low[MAXN], stk[MAXN], ins[MAXN], scc[MAXN];
int timer, top, scnt;

void tarjan(int u)
{
    dfn[u] = low[u] = ++timer;
    stk[++top] = u;
    ins[u] = 1;
    for (int v : g[u])
    {
        if (!dfn[v])
        {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        }
        else if (ins[v]) low[u] = min(low[u], dfn[v]);
    }
    if (low[u] == dfn[u])
    {
        scnt++;
        int v;
        do
        {
            v = stk[top--];
            ins[v] = 0;
            scc[v] = scnt;
        } while (v != u);
    }
}
```

使用说明：
- 对每个未访问点调用 `tarjan(i)`。
- 缩点后 `scc[u]` 为所属分量编号，遍历原边建新 DAG。
- 判定：`low[v] >= dfn[u]`（u 为割点条件之一，见模板 8）。

变体：
- 缩点 + 拓扑排序求 DAG 最长链 / 方案数。
- 2-SAT：矛盾关系建边，SCC 判断可行性（`x 与 !x 同 SCC` 则无解）。

## 8. Tarjan 割点 / 桥

适用：无向图求割点（删去后连通分量数增加）与桥（割边）。

复杂度：$O(n+m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
vector<int> g[MAXN];
int dfn[MAXN], low[MAXN], timer;
int pa[MAXN];
int cut[MAXN];

void tarjan_cut(int u)
{
    dfn[u] = low[u] = ++timer;
    int child = 0;
    for (int v : g[u])
    {
        if (v == pa[u]) continue;
        if (!dfn[v])
        {
            pa[v] = u;
            child++;
            tarjan_cut(v);
            low[u] = min(low[u], low[v]);
            if (pa[u] == 0 && child > 1) cut[u] = 1;
            if (pa[u] != 0 && low[v] >= dfn[u]) cut[u] = 1;
        }
        else low[u] = min(low[u], dfn[v]);
    }
}
```

使用说明：
- 根判断用 `pa[u] == 0`（根节点 pa 为 0），对每个未访问点调用。
- `cut[u]=1` 表示 u 是割点。重边不影响割点判定。

桥（割边）——用边编号判重：

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int head[MAXN], to[MAXN * 2], nxt[MAXN * 2], ecnt;
int dfn[MAXN], low[MAXN], timer;
int bridge[MAXN * 2];

void add_edge(int u, int v)
{
    to[++ecnt] = v;
    nxt[ecnt] = head[u];
    head[u] = ecnt;
}

void tarjan_bridge(int u, int fe)
{
    dfn[u] = low[u] = ++timer;
    for (int e = head[u]; e; e = nxt[e])
    {
        int v = to[e];
        if ((e ^ 1) == fe) continue;
        if (!dfn[v])
        {
            tarjan_bridge(v, e);
            low[u] = min(low[u], low[v]);
            if (low[v] > dfn[u]) bridge[e] = bridge[e ^ 1] = 1;
        }
        else low[u] = min(low[u], dfn[v]);
    }
}
```

使用说明：
- 无向边按 `add_edge(u,v); add_edge(v,u);` 成对加，`ecnt` 从 0 开始（`e^1` 是反向边）。
- 桥条件 `low[v] > dfn[u]`（不含等号，区别于割点）。
- 边双连通缩点：删桥后剩余连通块。

变体：
- 点双连通分量：栈维护边，`low[v] >= dfn[u]` 时弹出分量。
- 圆方树：点双建方点，可处理无向图任意两点路径问题。

## 9. Dinic 最大流

适用：网络最大流、二分图多重匹配、拆点建模（点容量）。

复杂度：$O(n^2m)$（一般图），二分图等特殊图更快。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1005;
const int MAXM = 200005;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
int head[MAXN], to[MAXM], nxt[MAXM], ecnt;
ll cap[MAXM];
int dis[MAXN], cur[MAXN];
int n, m, S, T;

void add_edge(int u, int v, ll c)
{
    to[++ecnt] = v;
    cap[ecnt] = c;
    nxt[ecnt] = head[u];
    head[u] = ecnt;
    to[++ecnt] = u;
    cap[ecnt] = 0;
    nxt[ecnt] = head[v];
    head[v] = ecnt;
}

bool bfs()
{
    memset(dis, -1, sizeof(dis));
    queue<int> q;
    q.push(S);
    dis[S] = 0;
    cur[S] = head[S];
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        for (int e = head[u]; e; e = nxt[e])
        {
            int v = to[e];
            if (cap[e] > 0 && dis[v] == -1)
            {
                dis[v] = dis[u] + 1;
                cur[v] = head[v];
                if (v == T) return true;
                q.push(v);
            }
        }
    }
    return false;
}

ll dfs(int u, ll f)
{
    if (u == T) return f;
    ll used = 0;
    for (int &e = cur[u]; e; e = nxt[e])
    {
        int v = to[e];
        if (cap[e] > 0 && dis[v] == dis[u] + 1)
        {
            ll d = dfs(v, min(f - used, cap[e]));
            if (d > 0)
            {
                cap[e] -= d;
                cap[e ^ 1] += d;
                used += d;
                if (used == f) break;
            }
        }
    }
    if (!used) dis[u] = -1;
    return used;
}

ll dinic()
{
    ll flow = 0;
    while (bfs())
    {
        flow += dfs(S, INF);
    }
    return flow;
}
```

使用说明：
- `ecnt` 从 0 开始，反向边用 `e^1` 自动配对。
- `S` 源点、`T` 汇点须在调用前设置，节点编号 1..n。
- 点容量：拆点 u → u'，中间连容量为点容量，其余边容量为 INF。

变体：
- 最大流最小割定理：最大流 = 最小割容量，`bfs` 后从 S 可达的点为割集一侧。
- 二分图最大匹配可建超级源汇跑 Dinic（比匈牙利稳）。
- 最大权闭合子图：正权连 S、负权连 T，答案 = 正权和 - 最小割。

## 10. 最小费用最大流（SPFA 版）

适用：每条边带费用的最大流，求最小总费用（如最小费用分配）。

复杂度：$O(Fnm)$（$F$ 为流量），数据小时够用；大流量换 Dinic+势能版。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1005;
const int MAXM = 200005;
const ll INF = 0x3f3f3f3f3f3f3f3fLL;
int head[MAXN], to[MAXM], nxt[MAXM], ecnt;
ll cap[MAXM], cost[MAXM];
ll dis[MAXN];
int pre[MAXN], pree[MAXN], inq[MAXN];
int n, m, S, T;

void add_edge(int u, int v, ll c, ll w)
{
    to[++ecnt] = v;
    cap[ecnt] = c;
    cost[ecnt] = w;
    nxt[ecnt] = head[u];
    head[u] = ecnt;
    to[++ecnt] = u;
    cap[ecnt] = 0;
    cost[ecnt] = -w;
    nxt[ecnt] = head[v];
    head[v] = ecnt;
}

bool spfa()
{
    for (int i = 1; i <= n; i++)
    {
        dis[i] = INF;
        inq[i] = 0;
        pre[i] = 0;
    }
    dis[S] = 0;
    queue<int> q;
    q.push(S);
    inq[S] = 1;
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        inq[u] = 0;
        for (int e = head[u]; e; e = nxt[e])
        {
            int v = to[e];
            if (cap[e] > 0 && dis[v] > dis[u] + cost[e])
            {
                dis[v] = dis[u] + cost[e];
                pre[v] = u;
                pree[v] = e;
                if (!inq[v])
                {
                    q.push(v);
                    inq[v] = 1;
                }
            }
        }
    }
    return dis[T] != INF;
}

pair<ll, ll> mcmf()
{
    ll flow = 0, mincost = 0;
    while (spfa())
    {
        ll f = INF;
        for (int v = T; v != S; v = pre[v]) f = min(f, cap[pree[v]]);
        for (int v = T; v != S; v = pre[v])
        {
            cap[pree[v]] -= f;
            cap[pree[v] ^ 1] += f;
        }
        flow += f;
        mincost += f * dis[T];
    }
    return {flow, mincost};
}
```

使用说明：
- 边成对加，反向边费用取负。`ecnt` 从 0 开始。
- 返回 `(最大流, 最小费用)`。
- 含负费用环时 SPFA 仍可跑（允许负边），但存在负权环会死循环，需预处理消环。

变体：
- 最大费用最大流：费用取负再跑，答案取负。
- 上 / 下界费用流：先处理下界（超级源汇），再跑。
- 流量限定：二分流量或加汇点边容量。

## 11. 匈牙利二分图最大匹配

适用：二分图最大匹配（n 较小），求最大匹配数 / 最小点覆盖 / 最大独立集。

复杂度：$O(nm)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 505;
vector<int> g[MAXN];
int match[MAXN];
int vis[MAXN];

bool dfs(int u)
{
    for (int v : g[u])
    {
        if (vis[v]) continue;
        vis[v] = 1;
        if (match[v] == 0 || dfs(match[v]))
        {
            match[v] = u;
            return true;
        }
    }
    return false;
}

int hungarian(int n)
{
    int res = 0;
    for (int i = 1; i <= n; i++)
    {
        memset(vis, 0, sizeof(vis));
        if (dfs(i)) res++;
    }
    return res;
}
```

使用说明：
- 左部点编号 1..n，右部点编号任意；`g[u]` 存左部点 u 可匹配的右部点。
- `match[v]` 记录右部 v 匹配的左部点。
- 二分图最大匹配 = 最小点覆盖 = n - 最大独立集（König 定理）。

变体：
- 多重匹配：右部拆点（每个右部点容量 k）。
- 带权匹配（Kuhn-Munkres）用小数据 / 费用流替代。

## 12. 拓扑排序

适用：DAG 判环、按依赖顺序处理任务、求 DAG 最长路 / 字典序最小拓扑序。

复杂度：$O(n+m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
vector<int> g[MAXN];
int indeg[MAXN];
int ans[MAXN];
int n, m, tot;

bool toposort()
{
    queue<int> q;
    for (int i = 1; i <= n; i++) if (indeg[i] == 0) q.push(i);
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        ans[++tot] = u;
        for (int v : g[u])
        {
            indeg[v]--;
            if (indeg[v] == 0) q.push(v);
        }
    }
    return tot == n;
}
```

使用说明：
- 返回 true 表示无环，`ans[1..n]` 为拓扑序；否则存在环。
- 字典序最小：把 queue 换成 priority_queue（小根堆）。

变体：
- 求 DAG 最长路：拓扑序上 DP，`dp[v] = max(dp[v], dp[u] + w)`。
- 反向拓扑可求"到达它的任务数"。

## 13. 差分约束系统

适用：形如 `x_v - x_u <= w` 的一组不等式，判断是否有解 / 求最大最小值。

复杂度：SPFA 判负环 $O(nm)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int INF = 0x3f3f3f3f;
vector<pair<int, int>> g[MAXN];
int dis[MAXN], cnt[MAXN], inq[MAXN];
int n, m;

void add_edge(int u, int v, int ww)
{
    g[u].push_back({v, ww});
}

bool diff_constraint()
{
    queue<int> q;
    for (int i = 1; i <= n; i++)
    {
        dis[i] = INF;
        cnt[i] = 0;
        inq[i] = 0;
    }
    for (int i = 1; i <= n; i++)
    {
        add_edge(0, i, 0);
        q.push(i);
        inq[i] = 1;
    }
    while (!q.empty())
    {
        int u = q.front();
        q.pop();
        inq[u] = 0;
        for (auto ed : g[u])
        {
            int v = ed.first;
            int w = ed.second;
            if (dis[v] > dis[u] + w)
            {
                dis[v] = dis[u] + w;
                if (!inq[v])
                {
                    q.push(v);
                    inq[v] = 1;
                    if (++cnt[v] > n + 1) return false;
                }
            }
        }
    }
    return true;
}
```

使用说明：
- 约束 `x_v - x_u <= w` 加边 u→v 权 w；`x_v - x_u >= w` 转成 u→v 权 -w。
- 建虚拟源点 0 连所有点权 0，保证全图连通。返回 false 表示无解（存在负环）。
- 求 `x_n - x_1` 最大值即 `dis[n]`（跑最短路）。

变体：
- 相等关系 `x_v - x_u = w`：拆成 `<= w` 和 `>= w` 两条。
- 严格小于 `x_v < x_u`：整数变量转 `x_v - x_u <= -1`。
