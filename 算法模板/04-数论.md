## 1. 快速幂 / 快速乘

适用：$O(\log n)$ 求 $a^b \bmod p$；快速乘防大数相乘溢出（模数 $\le 10^{18}$ 时）。

复杂度：$O(\log b)$，空间 $O(1)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll qpow(ll a, ll b, ll mod)
{
    ll res = 1;
    while (b)
    {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

ll qmul(ll a, ll b, ll mod)
{
    ll res = 0;
    while (b)
    {
        if (b & 1) res = (res + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
    }
    return res;
}
```

使用说明：
- 快速乘用于模数接近 $10^{18}$ 时，`a * b` 会溢出 ll 的场景。
- 模数很小时直接 `(a * b) % mod` 即可，不要滥用快速乘（慢）。

变体：
- 分数取模：$a/b \equiv a \cdot \mathrm{qpow}(b, p-2, p) \pmod p$（$p$ 为素数）。
- 矩阵快速幂见本文件模板 10。

## 2. 线性筛（素数 + 欧拉函数 + 莫比乌斯函数）

适用：一次筛出 n 以内所有素数，并顺带求欧拉函数 $\varphi$、莫比乌斯 $\mu$。

复杂度：$O(n)$，空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1000005;
int primes[MAXN], pcnt;
int phi[MAXN];
int mu[MAXN];
bool is_comp[MAXN];

void linear_sieve(int n)
{
    phi[1] = 1;
    mu[1] = 1;
    for (int i = 2; i <= n; i++)
    {
        if (!is_comp[i])
        {
            primes[++pcnt] = i;
            phi[i] = i - 1;
            mu[i] = -1;
        }
        for (int j = 1; j <= pcnt && i * primes[j] <= n; j++)
        {
            is_comp[i * primes[j]] = 1;
            if (i % primes[j] == 0)
            {
                phi[i * primes[j]] = phi[i] * primes[j];
                mu[i * primes[j]] = 0;
                break;
            }
            phi[i * primes[j]] = phi[i] * (primes[j] - 1);
            mu[i * primes[j]] = -mu[i];
        }
    }
}
```

使用说明：
- `primes[1..pcnt]` 为素数表；`phi[i]` 为欧拉函数（1..n 中与 i 互质个数）；`mu[i]` 为莫比乌斯函数。
- 欧拉定理：$\gcd(a,p)=1$ 时 $a^{\varphi(p)} \equiv 1 \pmod p$。
- 莫比乌斯反演：$[\gcd(i,j)=1] = \sum_{d\mid\gcd(i,j)} \mu(d)$。

变体：
- 只筛素数：去掉 phi/mu 数组，速度更快。
- 筛约数个数 / 约数和：加 `d[i]`、`cnt[i]`（最小质因子指数）。
- 杜教筛：n 到 1e10 时配合记忆化 + 分块。

## 3. 扩展欧几里得（exgcd）

适用：解 $ax+by=\gcd(a,b)$ 的一组整数解，或求乘法逆元、解一次同余方程。

复杂度：$O(\log \min(a,b))$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll exgcd(ll a, ll b, ll &x, ll &y)
{
    if (b == 0)
    {
        x = 1;
        y = 0;
        return a;
    }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}
```

使用说明：
- 返回值 $g=\gcd(a,b)$，$x,y$ 满足 $ax+by=g$。
- 解 $ax+by=c$：先求 $g$，$c$ 必须能被 $g$ 整除才有解；$x_0=x\cdot c/g,\ y_0=y\cdot c/g$。
- 通解：$x=x_0+k\cdot (b/g)$，$y=y_0-k\cdot (a/g)$。

变体：
- 求最小正整数解：对 $x_0$ 模 $b/g$ 取正。
- 解同余 $ax \equiv c \pmod b$：等价于 $ax+by=c$。

## 4. 乘法逆元

适用：mod p 下做除法 $a/b = a \cdot \mathrm{inv}(b)$。

复杂度：单次 $O(\log p)$（费马 / 扩欧），线性预处理 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1000005;
ll inv[MAXN];

ll qpow(ll a, ll b, ll mod)
{
    ll res = 1;
    while (b)
    {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

ll exgcd(ll a, ll b, ll &x, ll &y)
{
    if (b == 0)
    {
        x = 1;
        y = 0;
        return a;
    }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}

ll inv_fermat(ll a, ll mod)
{
    return qpow(a, mod - 2, mod);
}

ll inv_exgcd(ll a, ll mod)
{
    ll x, y;
    exgcd(a, mod, x, y);
    return (x % mod + mod) % mod;
}

void init_inv(int n, ll mod)
{
    inv[1] = 1;
    for (int i = 2; i <= n; i++) inv[i] = (mod - mod / i) * inv[mod % i] % mod;
}
```

使用说明：
- `inv_fermat` 要求 mod 为素数；`inv_exgcd` 只要求 $\gcd(a,\mathrm{mod})=1$。
- 线性预处理要求 mod 为素数且 i < mod。

变体：
- 预处理阶乘逆元（配合组合数）：见模板 6。
- 求 1..n 逆元前缀积可用于区间逆元。

## 5. 中国剩余定理 CRT / 扩展 CRT

适用：解同余方程组 $x \equiv a_i \pmod{m_i}$。CRT 要求 $m_i$ 两两互质，exCRT 无此要求。

复杂度：$O(n\log m)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll exgcd(ll a, ll b, ll &x, ll &y)
{
    if (b == 0)
    {
        x = 1;
        y = 0;
        return a;
    }
    ll g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}

ll qmul(ll a, ll b, ll mod)
{
    ll res = 0;
    while (b)
    {
        if (b & 1) res = (res + a) % mod;
        a = (a + a) % mod;
        b >>= 1;
    }
    return res;
}

ll inv_exgcd(ll a, ll mod)
{
    ll x, y;
    exgcd(a, mod, x, y);
    return (x % mod + mod) % mod;
}

ll crt(int n, ll *a, ll *m)
{
    ll M = 1;
    ll ans = 0;
    for (int i = 1; i <= n; i++) M *= m[i];
    for (int i = 1; i <= n; i++)
    {
        ll Mi = M / m[i];
        ll ti = inv_exgcd(Mi % m[i], m[i]);
        ans = (ans + qmul(qmul(a[i], Mi, M), ti, M)) % M;
    }
    return (ans % M + M) % M;
}

ll excrt(int n, ll *a, ll *m)
{
    ll M = m[1];
    ll ans = (a[1] % M + M) % M;
    for (int i = 2; i <= n; i++)
    {
        ll x, y;
        ll g = exgcd(M, m[i], x, y);
        ll c = ((a[i] - ans) % m[i] + m[i]) % m[i];
        if (c % g) return -1;
        ll t = qmul(x, c / g, m[i] / g);
        ans += qmul(t, M, M / g * m[i]);
        M = M / g * m[i];
        ans = (ans % M + M) % M;
    }
    return (ans % M + M) % M;
}
```

使用说明：
- 数组 a[1..n]、m[1..n] 存余数与模数。
- CRT 返回最小非负解；exCRT 无解返回 -1。
- 乘法都用 qmul 防溢出。

变体：
- 模数不互质直接用 exCRT（覆盖 CRT 全部场景）。

## 6. 组合数预处理 + Lucas

适用：$C(n,k)\bmod p$。$p$ 为素数且 $n$ 较小（$\le 10^6$）用阶乘逆元；$n$ 很大（$10^{18}$）用 Lucas。

复杂度：预处理 $O(n)$，查询 $O(1)$；Lucas 单次 $O(\log_p n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1000005;
ll fac[MAXN], inv_fac[MAXN];

ll qpow(ll a, ll b, ll mod)
{
    ll res = 1;
    while (b)
    {
        if (b & 1) res = res * a % mod;
        a = a * a % mod;
        b >>= 1;
    }
    return res;
}

void init_comb(int n, ll mod)
{
    fac[0] = 1;
    for (int i = 1; i <= n; i++) fac[i] = fac[i - 1] * i % mod;
    inv_fac[n] = qpow(fac[n], mod - 2, mod);
    for (int i = n - 1; i >= 0; i--) inv_fac[i] = inv_fac[i + 1] * (i + 1) % mod;
}

ll C(ll n, ll k, ll mod)
{
    if (k < 0 || k > n) return 0;
    return fac[n] * inv_fac[k] % mod * inv_fac[n - k] % mod;
}

ll lucas(ll n, ll k, ll mod)
{
    if (k == 0) return 1;
    return C(n % mod, k % mod, mod) * lucas(n / mod, k / mod, mod) % mod;
}
```

使用说明：
- 使用 C 前先 `init_comb(N, mod)`，$N \ge$ 所有 $n$。
- Lucas 用在 mod 较小、n 很大时；`C` 内部下标要 < mod。

变体：
- 卡特兰数：$\mathrm{Cat}(n) = C(2n,n)/(n+1)$（推导见 `trick合集.md` → 数学 → Raney 引理）。
- 组合数前缀和 / 隔板法（星条旗）模型：$C(n+k-1,k-1)$。

## 7. 数论分块（整除分块）

适用：形如 $\sum_{i=1}^{n}\lfloor n/i\rfloor$ 或带乘法的求和，$O(\sqrt n)$ 求值。

复杂度：$O(\sqrt n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll floor_sum(ll n)
{
    ll ans = 0;
    for (ll l = 1, r; l <= n; l = r + 1)
    {
        r = n / (n / l);
        ans += (r - l + 1) * (n / l);
    }
    return ans;
}
```

使用说明：
- 核心：$\lfloor n/i\rfloor$ 的值只有 $O(\sqrt n)$ 段，每段 $[l,r]$ 内值相同。
- 计算 $\sum \lfloor n/i\rfloor \cdot f(i)$ 时，$f(i)$ 用前缀和 $S(r)-S(l-1)$ 代替区间求和。

变体：
- 二维分块：$\sum\sum \lfloor n/i\rfloor \cdot \lfloor m/j\rfloor$，双指针 $l$ 取 $\min$ 分界。
- 配合莫比乌斯前缀和做数论求和。

## 8. 快速傅里叶变换 FFT

适用：两个多项式 / 数组卷积（$n,m\le 10^5$ 量级），乘出系数。涉及浮点精度。

复杂度：$O((n+m)\log(n+m))$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const double PI = acos(-1.0);
const int MAXN = 300005;
complex<double> f[MAXN], g[MAXN];
int rev[MAXN];

void fft(complex<double> *a, int n, int inv)
{
    for (int i = 0; i < n; i++) if (i < rev[i]) swap(a[i], a[rev[i]]);
    for (int len = 2; len <= n; len <<= 1)
    {
        complex<double> wn(cos(2 * PI / len), sin(2 * PI / len) * inv);
        for (int i = 0; i < n; i += len)
        {
            complex<double> w(1, 0);
            for (int j = 0; j < len / 2; j++)
            {
                complex<double> u = a[i + j];
                complex<double> v = w * a[i + j + len / 2];
                a[i + j] = u + v;
                a[i + j + len / 2] = u - v;
                w *= wn;
            }
        }
    }
    if (inv == -1) for (int i = 0; i < n; i++) a[i] /= n;
}

void multiply(int n, int m)
{
    int len = 1, bit = 0;
    while (len < n + m - 1)
    {
        len <<= 1;
        bit++;
    }
    for (int i = 1; i < len; i++) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (bit - 1));
    fft(f, len, 1);
    fft(g, len, 1);
    for (int i = 0; i < len; i++) f[i] *= g[i];
    fft(f, len, -1);
}
```

使用说明：
- 调用前把两个多项式系数分别放入 `f[0..n-1]`、`g[0..m-1]`。
- 结果系数存 `f[k]`（复部 ~0），取 `(ll)(f[k].real() + 0.5)`。
- 求大数乘积：拆成进制位做卷积再进位。

变体：
- 若模数支持 NTT，精度更稳，见模板 9。
- 卷积也可用于字符串匹配（错位匹配数）。

## 9. 快速数论变换 NTT

适用：模 998244353（原根 3）下的整数卷积，无浮点误差，适合模意义多项式。

复杂度：$O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const ll MOD = 998244353;
const ll G = 3;
const int MAXN = 300005;
ll a[MAXN], b[MAXN];
int rev[MAXN];

ll qpow(ll x, ll y)
{
    ll res = 1;
    while (y)
    {
        if (y & 1) res = res * x % MOD;
        x = x * x % MOD;
        y >>= 1;
    }
    return res;
}

void ntt(ll *p, int n, int inv)
{
    for (int i = 0; i < n; i++) if (i < rev[i]) swap(p[i], p[rev[i]]);
    for (int len = 2; len <= n; len <<= 1)
    {
        ll wn = qpow(G, (MOD - 1) / len);
        if (inv == -1) wn = qpow(wn, MOD - 2);
        for (int i = 0; i < n; i += len)
        {
            ll w = 1;
            for (int j = 0; j < len / 2; j++)
            {
                ll u = p[i + j];
                ll v = w * p[i + j + len / 2] % MOD;
                p[i + j] = (u + v) % MOD;
                p[i + j + len / 2] = (u - v + MOD) % MOD;
                w = w * wn % MOD;
            }
        }
    }
    if (inv == -1)
    {
        ll inv_n = qpow(n, MOD - 2);
        for (int i = 0; i < n; i++) p[i] = p[i] * inv_n % MOD;
    }
}

void multiply(int n, int m)
{
    int len = 1, bit = 0;
    while (len < n + m - 1)
    {
        len <<= 1;
        bit++;
    }
    for (int i = 1; i < len; i++) rev[i] = (rev[i >> 1] >> 1) | ((i & 1) << (bit - 1));
    ntt(a, len, 1);
    ntt(b, len, 1);
    for (int i = 0; i < len; i++) a[i] = a[i] * b[i] % MOD;
    ntt(a, len, -1);
}
```

使用说明：
- 系数放入 `a[0..n-1]`、`b[0..m-1]`，结果在 `a[0..n+m-2]`。
- 换模数需改 `MOD` 与 `G`（模数须为 $c\cdot 2^k+1$ 形式的 NTT 友好素数）。
- 大模数（$10^9+7$）用三模 NTT 合并。

变体：
- 多项式求逆 / 求导积分 / 求 ln、exp、开方：基于 NTT 的牛顿迭代（模板较长，赛中单独推）。
- 任意模数卷积：拆系数 FFT 或三模 NTT。

## 10. 矩阵快速幂（二维数组版）

适用：线性递推（斐波那契、常系数线性递推）加速到 $O(k^3\log n)$。

复杂度：$O(k^3\log n)$，空间 $O(k^2)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 105;
const ll MOD = 1000000007;
ll a[MAXN][MAXN];
ll res[MAXN][MAXN];
ll tmp[MAXN][MAXN];
int n;

void mul_res()
{
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            tmp[i][j] = 0;
            for (int k = 1; k <= n; k++) tmp[i][j] = (tmp[i][j] + res[i][k] * a[k][j]) % MOD;
        }
    }
    for (int i = 1; i <= n; i++) for (int j = 1; j <= n; j++) res[i][j] = tmp[i][j];
}

void mul_aa()
{
    for (int i = 1; i <= n; i++)
    {
        for (int j = 1; j <= n; j++)
        {
            tmp[i][j] = 0;
            for (int k = 1; k <= n; k++) tmp[i][j] = (tmp[i][j] + a[i][k] * a[k][j]) % MOD;
        }
    }
    for (int i = 1; i <= n; i++) for (int j = 1; j <= n; j++) a[i][j] = tmp[i][j];
}

void mat_qpow(ll b)
{
    for (int i = 1; i <= n; i++) for (int j = 1; j <= n; j++) res[i][j] = (i == j);
    while (b)
    {
        if (b & 1) mul_res();
        mul_aa();
        b >>= 1;
    }
}
```

使用说明：
- 把递推矩阵填入 `a[1..n][1..n]`，`n` 为矩阵阶数，调用 `mat_qpow(b)` 后 `res` 为 $a^b$。
- 斐波那契：$F_{k+2}=F_{k+1}+F_k$ 对应 2 阶矩阵 $\begin{smallmatrix}1&1\\1&0\end{smallmatrix}$，初始向量乘 $a^b$。
- 结果矩阵与初始向量相乘得递推第 $b$ 项。

变体：
- 改 MOD 即通用；n 大（k 阶递推）用 BM 线性递推 + 多项式取模优化。
- 矩阵快速幂可做图上长度为 k 的路径计数（邻接矩阵 k 次幂）。

## 11. 高斯消元（浮点方程组，列主元 + 判解）

适用：$n$ 元线性方程组 $Ax = b$ 求唯一解、判无解、判无穷多解；期望 DP 环上解方程、实数域线性基、矩阵求逆都基于它。

复杂度：$O(n^3)$。

约定：`a[i][1..n]` 为系数、`a[i][n+1]` 为右端项；`gauss()` 返回 1（唯一解，`x[1..n]` 有效）、0（无穷多解，只保证有解）、-1（无解）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 205;
const double EPS = 1e-8;
double a[MAXN][MAXN];
double x[MAXN];
int pcol[MAXN];
int n;

int gauss()
{
    int r = 1;
    for (int c = 1; c <= n; c++)
    {
        int p = r;
        for (int i = r; i <= n; i++) if (fabs(a[i][c]) > fabs(a[p][c])) p = i;
        if (fabs(a[p][c]) < EPS) continue;
        for (int j = c; j <= n + 1; j++) swap(a[r][j], a[p][j]);
        for (int j = n + 1; j >= c; j--) a[r][j] /= a[r][c];
        for (int i = 1; i <= n; i++)
        {
            if (i == r) continue;
            if (fabs(a[i][c]) < EPS) continue;
            for (int j = n + 1; j >= c; j--) a[i][j] -= a[i][c] * a[r][j];
        }
        pcol[r] = c;
        r++;
    }
    for (int i = r; i <= n; i++) if (fabs(a[i][n + 1]) > EPS) return -1;
    if (r <= n) return 0;
    for (int i = 1; i <= n; i++) x[pcol[i]] = a[i][n + 1];
    return 1;
}
```

使用说明：
- 读入后直接调 `gauss()`，判返回值再取 `x[]`。
- 自由元所在列被 `continue` 跳过（`pcol` 记录每个主元列），非满秩时 `pcol` 不覆盖全部列，故返回 0 时不给 `x[]`。
- 无穷多解要输出一组特解：把自由元全部置 0，再对每个主元回代 `x[pcol[i]] = a[i][n+1]`（此时它们已不含自由元部分）。

变体：
- 取模版：把 `EPS` 判断改成 `!a[p][c]`，除法换成乘逆元，即模意义高斯消元（解同余方程组）。
- 矩阵求逆：对 $[A \mid I]$ 做同样的行变换，左半变单位阵时右半即 $A^{-1}$。
- 精度敏感（系数数量级悬殊）时改用列主元 + 迭代改进，或直接上 `long double`。

## 12. 异或方程组 / 01 方程组高斯消元（bitset 加速）

适用：方程为"若干变量的异或 = 常数"（开关问题、灯阵、异或线性相关判定），或模 2 意义下的线性方程组。

复杂度：$O\dfrac{n^2 m}{64}$。

约定：`n` 个方程、`m` 个变量，`a[i][1..m]` 为系数、`a[i][m+1]` 为常数项；返回 1 唯一解、0 多解（`x[]` 给出一组特解）、-1 无解。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 505;
const int MAXB = 520;
bitset<MAXB> a[MAXN];
int pcol[MAXN], res[MAXN];
int n, m;

int gauss_xor()
{
    int r = 1;
    for (int c = 1; c <= m && r <= n; c++)
    {
        int p = 0;
        for (int i = r; i <= n; i++) if (a[i][c]) { p = i; break; }
        if (!p) continue;
        swap(a[r], a[p]);
        for (int i = 1; i <= n; i++) if (i != r && a[i][c]) a[i] ^= a[r];
        pcol[r] = c;
        r++;
    }
    for (int i = r; i <= n; i++) if (a[i][m + 1]) return -1;
    for (int i = 1; i <= r - 1; i++) res[pcol[i]] = a[i][m + 1];
    if (r <= m) return 0;
    return 1;
}
```

使用说明：
- 建方程用 `a[i][j] = 1` 直接置系数位，常数项写 `a[i][m+1]`。
- 返回 0（多解）时 `res[]` 是自由元取 0 的一组特解；要求异或值最小的解则对每个自由元尝试翻转（或用线性基）。
- 只要判"方程是否线性相关 / 最大无关组大小"，用线性基（见第 13 节）比消元更省。

变体：
- 求方案数：多解时答案为 $2^{m-r+1}$（自由元个数次幂）。
- 灯阵 / 翻棋子：每个格子的操作方程取邻域异或，未知数个数 = 格子数。

## 13. 线性基（异或）

适用：异或集合的最大值 / 最小值 / 第 k 小值、判断某数能否被异或表出、区间异或最值（前缀线性基）、图上最大 XOR 路径。

复杂度：单次插入 / 查询 $O(\log V)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int LOG = 62;
ll basis[LOG + 1];
ll d[LOG + 1];
int d_cnt;

void insert_ll(ll x)
{
    for (int i = LOG; i >= 0; i--)
    {
        if (!(x >> i & 1)) continue;
        if (!basis[i])
        {
            basis[i] = x;
            return;
        }
        x ^= basis[i];
    }
}

ll query_max()
{
    ll res = 0;
    for (int i = LOG; i >= 0; i--) if ((res ^ basis[i]) > res) res ^= basis[i];
    return res;
}

ll query_min()
{
    for (int i = 0; i <= LOG; i++) if (basis[i]) return basis[i];
    return 0;
}

void rebuild()
{
    for (int i = LOG; i >= 0; i--)
    {
        for (int j = i - 1; j >= 0; j--) if (basis[i] >> j & 1) basis[i] ^= basis[j];
    }
    d_cnt = 0;
    for (int i = 0; i <= LOG; i++) if (basis[i]) d[d_cnt++] = basis[i];
}

ll query_kth(ll k, bool has_zero)
{
    if (has_zero) k--;
    if (k <= 0) return 0;
    if (k >= (1LL << d_cnt)) return -1;
    ll res = 0;
    for (int i = 0; i < d_cnt; i++) if (k >> i & 1) res ^= d[i];
    return res;
}
```

使用说明：
- `LOG` 取数的二进制位数上限：`ll` 用 62，`int` 用 30。
- `query_kth` 前必须先 `rebuild()`；`has_zero` 表示原集合能否异或出 0（即插入时有元素被消成 0）。
- `query_min()` 返回最小非零值，集合含 0 时最小值为 0。

变体：
- 前缀线性基（区间 $[l,r]$ 异或最值）：每个位置存一份基并记录每个基向量的最大插入位置，查询时只取位置 $\ge l$ 的向量。
- 判断 $x$ 能否表出：`insert_ll` 的循环中若 $x$ 被消成 0 则可表出。
- 求秩（最大无关组大小）：插入成功的次数。

## 14. 莫比乌斯反演：gcd 计数

适用：$\sum\sum f(\gcd(i,j))$ 型计数（互质对个数、$\gcd$ 求和、LCM 求和），把按 $\gcd$ 求和转成按约数求和再整除分块。

复杂度：筛 $O(n)$，单次询问 $O(\sqrt n)$。

约定：`get_mu(n)` 先跑，得到 `mu[]` 与其前缀和 `smu[]`；`solve(n,m)` 求 $\sum_{i=1}^{n}\sum_{j=1}^{m}[\gcd(i,j)=1]$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1000005;
int mu[MAXN];
ll smu[MAXN];
int primes[MAXN], pcnt;
bool vis[MAXN];

void get_mu(int n)
{
    mu[1] = 1;
    for (int i = 2; i <= n; i++)
    {
        if (!vis[i])
        {
            primes[++pcnt] = i;
            mu[i] = -1;
        }
        for (int j = 1; j <= pcnt && i * primes[j] <= n; j++)
        {
            vis[i * primes[j]] = true;
            if (i % primes[j] == 0)
            {
                mu[i * primes[j]] = 0;
                break;
            }
            mu[i * primes[j]] = -mu[i];
        }
    }
    for (int i = 1; i <= n; i++) smu[i] = smu[i - 1] + mu[i];
}

ll solve(int n, int m)
{
    if (n > m) swap(n, m);
    ll res = 0;
    for (int l = 1, r; l <= n; l = r + 1)
    {
        r = min(n / (n / l), m / (m / l));
        res += (smu[r] - smu[l - 1]) * (ll)(n / l) * (ll)(m / l);
    }
    return res;
}
```

使用说明：
- 反演依据：$[\gcd(i,j)=1] = \sum_{d \mid \gcd(i,j)} \mu(d)$，交换求和序后内层变为 $\left\lfloor n/d \right\rfloor\left\lfloor m/d \right\rfloor$。
- 求 $\sum_{i,j} \gcd(i,j)$ 则用 $f(d)=d$ 的倍数形式，或 $\varphi$ 前缀和：$\sum_{d}\varphi(d)\left\lfloor n/d\right\rfloor\left\lfloor m/d\right\rfloor$。
- 单次询问 $n \le 10^5$ 时可只跑到 $n$ 不化简，多组询问务必整除分块。

变体：
- 上下界 $[a,b]\times[c,d]$：二维前缀和容斥，`solve(b,d)-solve(a-1,d)-solve(b,c-1)+solve(a-1,c-1)`。
- 带权 $f(\gcd)$：预处理 $g(d)=\sum_{k\mid d}\mu(k)f(d/k)$（狄利克雷卷积），再整除分块。
- 杜教筛：$n \ge 10^7$ 且仍要前缀和时用 $\sum\mu$ / $\sum\varphi$ 的杜教筛，$O(n^{2/3})$。

## 15. 扩展 Lucas（exLucas）

适用：求 $\binom{n}{m} \bmod P$，而 $P$ 不保证是质数（也不保证与 $n$ 互质），如 $P = 999911659$ 这类合数模。

复杂度：$O(P^{1/2}\log P)$ 量级（对模数的每个素因子幂处理）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define i128 __int128

ll qpow(ll a, ll b, ll p)
{
    ll res = 1 % p;
    a %= p;
    while (b)
    {
        if (b & 1) res = (i128)res * a % p;
        a = (i128)a * a % p;
        b >>= 1;
    }
    return res;
}

void exgcd(ll a, ll b, ll &x, ll &y)
{
    if (!b)
    {
        x = 1;
        y = 0;
        return;
    }
    exgcd(b, a % b, y, x);
    y -= a / b * x;
}

ll inv(ll a, ll p)
{
    ll x, y;
    exgcd(a, p, x, y);
    return (x % p + p) % p;
}

ll calc_e(ll n, ll p)
{
    ll res = 0;
    while (n)
    {
        res += n / p;
        n /= p;
    }
    return res;
}

ll fac_mod(ll n, ll p, ll pk)
{
    if (!n) return 1;
    ll res = 1;
    for (ll i = 2; i <= pk; i++) if (i % p) res = (i128)res * i % pk;
    res = qpow(res, n / pk, pk);
    for (ll i = 2; i <= n % pk; i++) if (i % p) res = (i128)res * i % pk;
    return (i128)res * fac_mod(n / p, p, pk) % pk;
}

ll C_mod_pk(ll n, ll m, ll p, ll pk)
{
    ll e = calc_e(n, p) - calc_e(m, p) - calc_e(n - m, p);
    ll res = fac_mod(n, p, pk);
    res = (i128)res * inv(fac_mod(m, p, pk), pk) % pk;
    res = (i128)res * inv(fac_mod(n - m, p, pk), pk) % pk;
    return (i128)res * qpow(p, e, pk) % pk;
}

ll exlucas(ll n, ll m, ll P)
{
    if (m < 0 || m > n) return 0;
    ll res = 0, rest = P;
    for (ll p = 2; p * p <= rest; p++)
    {
        if (rest % p) continue;
        ll pk = 1;
        while (rest % p == 0)
        {
            pk *= p;
            rest /= p;
        }
        ll M = P / pk;
        ll part = (i128)C_mod_pk(n, m, p, pk) * (M % P) % P;
        res = (res + (i128)part * inv(M % pk, pk)) % P;
    }
    if (rest > 1)
    {
        ll p = rest, pk = rest;
        ll M = P / pk;
        ll part = (i128)C_mod_pk(n, m, p, pk) * (M % P) % P;
        res = (res + (i128)part * inv(M % pk, pk)) % P;
    }
    return res;
}
```

使用说明：
- 调用 `exlucas(n, m, P)` 直接返回 $\binom{n}{m} \bmod P$，$P$ 可为任意正整数（$P=1$ 返回 0）。
- 思路：把 $P$ 分解为 $\prod p_i^{k_i}$，在各 $p^k$ 下算出 $C$ 的值（阶乘除去 $p$ 因子后递归），再用 CRT 合并。
- 所有乘法用 `__int128` 中转，防 $P \le 10^9$ 以上时中间积溢出。

变体：
- $P$ 是质数时直接用 Lucas（见第 6 节），不要绕 exLucas。
- 需要多组询问同一 $P$：把 $P$ 的分解结果缓存起来，避免重复分解。

## 16. 离散对数 BSGS

适用：解 $a^x \equiv b \pmod p$（$\gcd(a,p)=1$），如取模下的对数、NTT 之外的原根幂次定位。

复杂度：$O(\sqrt p)$（哈希表常数）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define i128 __int128

ll qpow(ll a, ll b, ll p)
{
    ll res = 1 % p;
    a %= p;
    while (b)
    {
        if (b & 1) res = (i128)res * a % p;
        a = (i128)a * a % p;
        b >>= 1;
    }
    return res;
}

ll bsgs(ll a, ll b, ll p)
{
    a %= p;
    b %= p;
    if (p == 1) return 0;
    if (b == 1) return 0;
    ll m = (ll)ceil(sqrt((double)p));
    unordered_map<ll, ll> mp;
    mp.reserve(m * 2);
    ll cur = b;
    for (ll i = 0; i < m; i++)
    {
        if (!mp.count(cur)) mp[cur] = i;
        cur = (i128)cur * a % p;
    }
    ll am = qpow(a, m, p);
    cur = 1;
    for (ll i = 1; i <= m; i++)
    {
        cur = (i128)cur * am % p;
        if (mp.count(cur)) return i * m - mp[cur];
    }
    return -1;
}
```

使用说明：
- 返回最小非负解，无解返回 -1；$x$ 满足 $a^x \equiv b$，$b=1$ 时解为 0。
- 要求 $\gcd(a,p)=1$，否则先做扩展 BSGS（每步把 $a$ 与 $p$ 的 $\gcd$ 提出来单独记）。
- 块长取 $\sqrt p$，复杂度 $O(\sqrt p)$，$p \le 10^{12}$ 可用。

变体：
- 扩展 BSGS：$g=\gcd(a,p)>1$ 时把方程改成 $a^{x-1}\cdot(a/g) \equiv b/g \pmod{p/g}$ 迭代约 $\log p$ 次再归约到互质情形。
- 求原根的指标（离散对数表）：对质数 $p$ 取原根 $g$ 后，$\mathrm{ind}(x)=\text{bsgs}(g,x,p)$。

## 17. 原根

适用：求质数（或 $2,4,p^k,2p^k$）的最小原根，NTT 中确定单位根、判定阶与循环群结构。

复杂度：分解 $p-1$ 为 $O(\sqrt p)$，枚举验证 $O(g\log p)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define i128 __int128

ll qpow(ll a, ll b, ll p)
{
    ll res = 1 % p;
    a %= p;
    while (b)
    {
        if (b & 1) res = (i128)res * a % p;
        a = (i128)a * a % p;
        b >>= 1;
    }
    return res;
}

ll fac[105];
int fcnt;

bool is_primitive_root(ll g, ll p)
{
    if (qpow(g, p - 1, p) != 1) return false;
    for (int i = 1; i <= fcnt; i++) if (qpow(g, (p - 1) / fac[i], p) == 1) return false;
    return true;
}

ll primitive_root(ll p)
{
    if (p == 2) return 1;
    ll t = p - 1;
    fcnt = 0;
    for (ll i = 2; i * i <= t; i++)
    {
        if (t % i) continue;
        fac[++fcnt] = i;
        while (t % i == 0) t /= i;
    }
    if (t > 1) fac[++fcnt] = t;
    for (ll g = 2; g < p; g++) if (is_primitive_root(g, p)) return g;
    return -1;
}
```

使用说明：
- 仅 $p = 2, 4, q^k, 2q^k$（$q$ 为奇素数）存在原根，其余返回 -1；竞赛里主要是质数模。
- $g$ 是原根 $\iff g^{p-1}\equiv 1$ 且对 $p-1$ 的每个素因子 $q$ 有 $g^{(p-1)/q} \not\equiv 1$。
- 求阶：$\mathrm{ord}(a) = \min d \mid p-1$ 使 $a^d\equiv1$，可对 $p-1$ 的素因子逐个试除$d$ 得到。

变体：
- 最小原根一般很小（$\le p^{0.25}$ 量级），从 2 枚举不会慢。
- 求所有原根：$g^k$（$\gcd(k,p-1)=1$）全是原根，个数为 $\varphi(p-1)$。
- 需要 $p-1$ 的分解且 $p$ 很大时配 Miller–Rabin + Pollard–Rho（见 `10-随机化.md`）。

## 18. 置换计数：Burnside 引理 / Pólya 定理

适用：本质不同染色方案计数（项链、手串、正方体染色、旋转 + 翻转等价类），"在群作用下数轨道数"。

复杂度：$O(|G| \cdot \log)$（枚举群元素）。

约定：`c` 为颜色数，`MOD` 为模数；`polya_rotation(n, c)` 只考虑旋转，`polya_bracelet(n, c)` 旋转 + 翻转（手链）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const ll MOD = 1000000007;

ll qpow(ll a, ll b, ll p)
{
    ll res = 1 % p;
    a %= p;
    while (b)
    {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

ll inv(ll a, ll p)
{
    return qpow(a, p - 2, p);
}

ll polya_rotation(int n, ll c)
{
    ll res = 0;
    for (int i = 1; i <= n; i++) res = (res + qpow(c, __gcd(n, i), MOD)) % MOD;
    return res % MOD * inv(n % MOD, MOD) % MOD;
}

ll polya_bracelet(int n, ll c)
{
    ll res = 0;
    for (int i = 1; i <= n; i++) res = (res + qpow(c, __gcd(n, i), MOD)) % MOD;
    if (n & 1) res = (res + (ll)n % MOD * qpow(c, (n + 1) / 2, MOD)) % MOD;
    else res = (res + (ll)(n / 2) % MOD * qpow(c, n / 2, MOD) + (ll)(n / 2) % MOD * qpow(c, n / 2 + 1, MOD)) % MOD;
    return res % MOD * inv(2 * (ll)n % MOD, MOD) % MOD;
}
```

使用说明：
- Burnside：轨道数 $= \dfrac{1}{|G|}\sum_{g\in G} |\mathrm{Fix}(g)|$；置换 $g$ 的不动点数是 $c^{(\text{循环数})}$。
- 旋转置换 $i$ 步的循环数为 $\gcd(n,i)$；$n$ 元循环的循环数即 $\gcd$ 结果。
- 手链（允许翻转）：奇数 $n$ 有 $n$ 个翻转置换各贡献 $c^{(n+1)/2}$；偶数 $n$ 分过两珠、过两缝两类各 $n/2$ 个，指数分别是 $n/2+1$ 与 $n/2$。

变体：
- 大 $n$（$10^9$）时把 $\sum_{d\mid n}\varphi(d)c^{n/d}$ 按约数枚举，需分解 $n$ 求约数与 $\varphi$。
- 正方体面染色：$G$ 为 24 阶旋转群，按"1 个单位元（6 循环）、6 个 90°（3）、3 个 180°（4）、8 个 120°（2）、6 个 180° 对棱（3）"求和。
- 颜色不自由（各颜色有数量限制）时 Pólya 失效，改用 Burnside + 生成函数 / DP。

## 19. 自适应 Simpson 积分

适用：求 $\int_a^b f(x)\,dx$，$f$ 光滑但无初等原函数（圆与多边形交面积、期望连续化、面积并）。

复杂度：$O(\text{递归层数})$，近似 $O(\log \frac{1}{\varepsilon})$。

约定：把被积函数写成 `f(x)`，调用 `integral(l, r, eps)` 返回积分值。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

double f(double x)
{
    return sin(x) / x;
}

double simpson(double l, double r)
{
    double mid = (l + r) / 2;
    return (r - l) * (f(l) + 4 * f(mid) + f(r)) / 6;
}

double asr(double l, double r, double eps, double S)
{
    double mid = (l + r) / 2;
    double sl = simpson(l, mid), sr = simpson(mid, r);
    if (fabs(sl + sr - S) <= 15 * eps) return sl + sr + (sl + sr - S) / 15;
    return asr(l, mid, eps / 2, sl) + asr(mid, r, eps / 2, sr);
}

double integral(double l, double r, double eps)
{
    return asr(l, r, eps, simpson(l, r));
}
```

使用说明：
- `eps` 取 $10^{-8}$ 左右即可，`f` 换成实际被积函数（注意 `f` 是常见函数名，与其他模板合并时改名）。
- 端点奇异（如 $x=0$ 处被积函数无定义）时把区间端点微移 $\varepsilon$，或换变量。
- 递归收敛判据用的是 Simpson 公式的误差主项 $\dfrac{1}{15}(S_{\text{粗}} - S_{\text{细}})$。

变体：
- 圆面积并：对 $x$ 积分"竖直线与各圆的交区间并长"，或直接用圆面积并的扫描线（见 `06-计算几何.md`）。
- 双重积分：外层 Simpson + 内层 Simpson 嵌套，常数约为 $O(\text{层数}^2)$。
- 需要高精度（$10^{-12}$）时把 `eps` 调小并注意递归深度，可用 `long double` 重写。

## 20. 拉格朗日插值

适用：已知 $n+1$ 个互异点 $(x_i, y_i)$，求多项式在 $x = k$ 处的值（模素数 p）；或已知答案关于 n 是 k 次多项式，用前 k+1 项推任意项。计数题"猜公式"的标准工具。

复杂度：$O(n^2)$；x 为连续整数时可 $O(n)$。

约定：点存 `px[0..n], py[0..n]`（全局），p 为素数。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 5005;
ll px[MAXN], py[MAXN];

ll qpow(ll a, ll b, ll p)
{
    ll res = 1;
    a %= p;
    while (b)
    {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

ll lagrange(int n, ll k, ll p)
{
    ll res = 0;
    for (int i = 0; i <= n; i++)
    {
        ll s = py[i] % p, t = 1;
        for (int j = 0; j <= n; j++)
        {
            if (j == i) continue;
            s = s * ((k - px[j]) % p + p) % p;
            t = t * ((px[i] - px[j]) % p + p) % p;
        }
        res = (res + s * qpow(t, p - 2, p)) % p;
    }
    return res;
}
```

使用说明：
- 调用前把 `k` 与所有 `x_i` 都对 p 取模，避免减法溢出。
- k 次多项式需要 k+1 个点；常用手法：暴力算 $n = 1..k+1$ 的答案，插值出 $n$ 更大的值。
- 若 $x_i$ 恰为 $0..n$（连续整数），用前缀积 / 后缀积可做到 $O(n)$：$f(k)=\sum_i y_i \dfrac{\text{pre}_{i-1}\,\text{suf}_{i+1}}{i!\,(n-i)!}$（pre = $\prod_{j<i}(k-j)$，suf = $\prod_{j>i}(k-j)$，分母阶乘逆元线性预处理）。

变体：
- 重心插值（点连续时 $O(n)$ 预处理单点查询）：维护 $\dfrac{1}{x_i - k}$ 两侧前缀后缀积。
- 求多项式系数本身用高斯消元 / 多项式插值的分治版本（赛场少见，直接用逐点值更常见）。

## 21. Prüfer 序列

适用：标号树与序列的双射。树计数（Cayley 公式、度数限制计数）、"给 n 个点数树"类问题。$O(n \log n)$（堆版，n 大时够用）。

约定：树以 `fa[i]`（i = 2..n，根为 1，`fa[1]` 无用）给出；Prüfer 序列存 `pru[1..n-2]`。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 500005;
int n, fa[MAXN], pru[MAXN], deg[MAXN];

void tree2prufer()
{
    for (int i = 1; i <= n; i++) deg[i] = 1;
    for (int i = 2; i <= n; i++) deg[fa[i]]++;
    priority_queue<int, vector<int>, greater<int>> q;
    for (int i = 1; i <= n; i++)
        if (deg[i] == 1) q.push(i);
    for (int i = 1; i <= n - 2; i++)
    {
        int u = q.top();
        q.pop();
        deg[u] = 0;
        pru[i] = fa[u];
        if (--deg[pru[i]] == 1) q.push(pru[i]);
    }
}

void prufer2tree()
{
    for (int i = 1; i <= n; i++) deg[i] = 1;
    for (int i = 1; i <= n - 2; i++) deg[pru[i]]++;
    priority_queue<int, vector<int>, greater<int>> q;
    for (int i = 1; i <= n; i++)
        if (deg[i] == 1) q.push(i);
    for (int i = 1; i <= n - 2; i++)
    {
        int u = q.top();
        q.pop();
        fa[u] = pru[i];
        if (--deg[pru[i]] == 1) q.push(pru[i]);
    }
    int a = q.top();
    q.pop();
    fa[a] = q.top();
}
```

使用说明：
- 性质：度数为 $d_i$ 的点在 Prüfer 序列中出现 $d_i - 1$ 次；序列长 $n-2$。
- Cayley 公式：$n$ 个点的标号生成树共 $n^{n-2}$ 棵。
- 指定度数 $d_1..d_n$ 的树数：$\dfrac{(n-2)!}{\prod_i (d_i-1)!}$（多重排列）。
- `prufer2tree` 还原后根是编号最大的剩余点；两函数互逆，可用来对拍。

变体：
- $O(n)$ 双指针版：堆换两指针，赛场通常堆版足够。
- 森林计数（指定 k 棵树的点集大小 $s_1..s_k$）：$k \cdot n^{k-2} \cdot \prod s_i$。
- 边带权 / 加约束的树计数一般转矩阵树定理（04 第 22 节）。

## 22. 矩阵树定理（生成树计数）

适用：无向图生成树计数（模素数 p）；有向图树形图计数（配合 BEST 定理数欧拉回路）。$O(n^3)$。

约定：边表 `eu[1..m], ev[1..m]`（无向）；`n` 为点数。结果 = 生成树个数 mod p。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 305;
int n;
ll a[MAXN][MAXN], d[MAXN];

ll qpow(ll a, ll b, ll p)
{
    ll res = 1;
    a %= p;
    while (b)
    {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

ll det_mod(int n, ll p)
{
    ll res = 1;
    for (int i = 1; i <= n; i++)
    {
        int r = 0;
        for (int j = i; j <= n; j++)
            if (a[j][i])
            {
                r = j;
                break;
            }
        if (!r) return 0;
        if (r != i)
        {
            for (int k = i; k <= n; k++) swap(a[i][k], a[r][k]);
            res = p - res;
        }
        res = res * a[i][i] % p;
        ll invv = qpow(a[i][i], p - 2, p);
        for (int j = i + 1; j <= n; j++)
        {
            ll f = a[j][i] * invv % p;
            if (f)
                for (int k = i; k <= n; k++)
                    a[j][k] = (a[j][k] - f * a[i][k] % p + p) % p;
        }
    }
    return res;
}

ll count_spanning(int m, ll p, int *eu, int *ev)
{
    memset(d, 0, sizeof(d));
    memset(a, 0, sizeof(a));
    for (int i = 1; i <= m; i++)
    {
        d[eu[i]]++;
        d[ev[i]]++;
        a[eu[i]][ev[i]] = (a[eu[i]][ev[i]] + p - 1) % p;
        a[ev[i]][eu[i]] = (a[ev[i]][eu[i]] + p - 1) % p;
    }
    for (int i = 1; i <= n; i++) a[i][i] = d[i] % p;
    return det_mod(n - 1, p);
}
```

使用说明：
- 基尔霍夫矩阵 = 度数矩阵 − 邻接矩阵，**去掉任意一行一列**（代码里去第 n 行列）求行列式。
- 重边算多条（对计数有贡献），自环直接忽略。
- p 为素数（用费马逆元）；模数非素数时改用辗转相除消元（变体）。

变体：
- BEST 定理：欧拉图（每个点入度 = 出度）的有向欧拉回路条数 = $t_w \cdot \prod_v (\text{outdeg}(v)-1)!$，其中 $t_w$ 是以 w 为根的**内向**树形图数（把每条边视为"子 → 父"再用矩阵树定理，去掉 w 所在行列）。
- 不取模的小图（n ≤ 10）可用 `long double` 高斯消元求行列式再四舍五入。
- 有向树形图计数：外向 / 内向分别对应"父 − 邻接"或"父 + 邻接"（转移方向取反）。

## 23. 杜教筛

适用：$n \le 10^{10}$ 左右求积性函数前缀和：$\sum_{i=1}^n \varphi(i)$、$\sum_{i=1}^n \mu(i)$，以及可配出 $g$ 的卷积（$\sum (f * g)(i)$ 好求时）。

复杂度：$O(n^{2/3})$（线性筛到 $n^{2/3}$）。

约定：先 `sieve(MAXV)`；`Sphi(x)` / `Smu(x)` 返回前缀和；`MAXV` 按 $n^{2/3}$ 取（$n \le 10^{10}$ 取 2e6~5e6，注意两个数组各 8 字节 / 个的内存）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXV = 2000000;
int vis[MAXV + 5], pr[MAXV / 10 + 5], pc;
ll phis[MAXV + 5], mus[MAXV + 5];
unordered_map<ll, ll> mphi, mmu;

void sieve(int n)
{
    phis[1] = 1;
    mus[1] = 1;
    for (int i = 2; i <= n; i++)
    {
        if (!vis[i])
        {
            pr[++pc] = i;
            phis[i] = i - 1;
            mus[i] = -1;
        }
        for (int j = 1; j <= pc && (ll)i * pr[j] <= n; j++)
        {
            vis[i * pr[j]] = 1;
            if (i % pr[j] == 0)
            {
                phis[i * pr[j]] = phis[i] * pr[j];
                mus[i * pr[j]] = 0;
                break;
            }
            phis[i * pr[j]] = phis[i] * (pr[j] - 1);
            mus[i * pr[j]] = -mus[i];
        }
    }
    for (int i = 1; i <= n; i++)
    {
        phis[i] += phis[i - 1];
        mus[i] += mus[i - 1];
    }
}

ll Sphi(ll x)
{
    if (x <= MAXV) return phis[x];
    if (mphi.count(x)) return mphi[x];
    ll res = x * (x + 1) / 2;
    for (ll l = 2, r; l <= x; l = r + 1)
    {
        r = x / (x / l);
        res -= (r - l + 1) * Sphi(x / l);
    }
    return mphi[x] = res;
}

ll Smu(ll x)
{
    if (x <= MAXV) return mus[x];
    if (mmu.count(x)) return mmu[x];
    ll res = 1;
    for (ll l = 2, r; l <= x; l = r + 1)
    {
        r = x / (x / l);
        res -= (r - l + 1) * Smu(x / l);
    }
    return mmu[x] = res;
}
```

使用说明：
- 原理一句话：$S_f(n) = \sum_{i=1}^n (f*g)(i) - \sum_{i=2}^n g(i) S_f(\lfloor n/i \rfloor)$，配合数论分块（04 第 7 节）。
- $\varphi$ 的 $g$ 是恒等函数（前缀和 $x(x+1)/2$）；$\mu$ 的 $g$ 是常数 1。
- 新的积性函数 f：找到 g 使 $f*g$ 前缀和可 $O(1)$ / $O(\log)$ 算，套同样递归骨架。
- 递归层数深，`unordered_map` 记忆化必要；卡哈希时换自定义哈希（09 附录）。

变体：
- Min_25 筛：更强的积性函数前缀和（质数位置的函数值可用多项式 / 数论函数描述），赛场极少手写。
- 狄利克雷前缀和 / 快速调和级数求 $\sum \lfloor n/i \rfloor$ 型都用数论分块即可。

## 24. 类欧几里得（floor_sum）

适用：$\sum_{i=0}^{n} \left\lfloor \dfrac{ai+b}{c} \right\rfloor$ 精确求值；分数下取整和、交换求和顺序后产生的余数和。

复杂度：$O(\log)$，与欧几里得算法同型。

约定：$n \ge 0$，$c \ge 1$，$a, b \ge 0$；返回精确值（可能很大，注意溢出，见说明）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll fsum(ll n, ll a, ll b, ll c)
{
    if (a == 0) return (b / c) * (n + 1);
    if (a >= c || b >= c)
        return (a / c) * n * (n + 1) / 2 + (b / c) * (n + 1) + fsum(n, a % c, b % c, c);
    ll m = (ll)((__int128)a * n + b) / c;
    if (m == 0) return 0;
    return m * (n + 1) - fsum(m - 1, c, c + a - 1 - b, a);
}
```

使用说明：
- $a, b$ 非负；$a$ 可能为负时先化正：$\lfloor (ai+b)/c \rfloor$ 按 $i$ 奇偶 / 补常数拆开，或写带符号版。
- 溢出边界：`(a/c) * n * (n+1) / 2` 在 $a, n \le 10^9$ 量级内安全；更大值域把相关乘法换 `__int128`。
- 求和下标从 0 开始；题目从 1 开始就传 $n-1$ 或结果做差。

变体：
- $\sum i \lfloor (ai+b)/c \rfloor$、$\sum \lfloor (ai+b)/c \rfloor^2$：同型递归多带两项，赛场需要时按推导补。
- 万能欧几里得：把矩阵挂在欧几里得过程上，一次处理多个下取整和的组合（难，先用上述基本款）。

## 25. 二次剩余（Cipolla）

适用：模素数 p 下解 $x^2 \equiv n \pmod p$；开根、模平方根。

复杂度：期望 $O(\log p)$。

约定：p 为奇素数（p = 2 直接返回 n mod 2）；返回任意一个根，另一个根是 `p - res`；无解返回 -1。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

ll W;

ll qpow(ll a, ll b, ll p)
{
    ll res = 1;
    a %= p;
    while (b)
    {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

struct Cnum
{
    ll x, y;
};

Cnum cmul(Cnum a, Cnum b, ll p)
{
    return Cnum{(a.x * b.x % p + a.y * b.y % p * W % p) % p, (a.x * b.y + a.y * b.x) % p};
}

ll sqrt_mod(ll n, ll p)
{
    n %= p;
    if (n < 0) n += p;
    if (n == 0) return 0;
    if (p == 2) return n;
    if (qpow(n, (p - 1) / 2, p) == p - 1) return -1;
    mt19937_64 rnd(chrono::steady_clock::now().time_since_epoch().count());
    ll a;
    while (true)
    {
        a = rnd() % p;
        W = (a * a % p - n + p) % p;
        if (W && qpow(W, (p - 1) / 2, p) == p - 1) break;
    }
    Cnum res{1, 0}, base{a, 1};
    ll e = (p + 1) / 2;
    while (e)
    {
        if (e & 1) res = cmul(res, base, p);
        base = cmul(base, base, p);
        e >>= 1;
    }
    return res.x;
}
```

使用说明：
- 判无解用欧拉判据：$n^{(p-1)/2} \equiv -1$。
- `W` 是"虚单位"的平方（$a^2 - n$ 是非二次剩余）；虚数域 $\mathbb{F}_{p^2}$ 上快速幂，结果落在实部。
- 同余方程 $x^2 \equiv n \pmod {p^k}$：先解 mod p，再 Hensel 提升或直接 BSGS 级别暴力（k 小）。

变体：
- 多项式开根（NTT 场景）：逐位 Hensel 提升，赛场少见。
- 合数模数：分解质因数后按 CRT 合并各素数幂的解。
