## 1. SG 函数（公平组合博弈）

适用：多堆 / 多局组合博弈，判断先手胜负（Nim 类、图上走棋子等）。

复杂度：$O(n \times \text{移动数})$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 1005;
int sg[MAXN];
bool vis[MAXN];
int f[MAXN];
int fn;

int get_sg(int x)
{
    if (sg[x] != -1) return sg[x];
    memset(vis, 0, sizeof(vis));
    for (int i = 1; i <= fn; i++) if (x >= f[i]) vis[get_sg(x - f[i])] = 1;
    for (int i = 0;; i++) if (!vis[i]) return sg[x] = i;
}

int xor_sum(int *a, int n)
{
    int res = 0;
    for (int i = 1; i <= n; i++) res ^= get_sg(a[i]);
    return res;
}
```

使用说明：
- `f[1..fn]` 为一次可取的步数集合，先 `memset(sg, -1, sizeof(sg))`。
- 单局 SG 值异或和 > 0 则先手必胜，= 0 必败。
- 求必胜策略（取走哪堆多少个）：枚举每堆取走后异或和为 0。

变体：
- 反 Nim / 有向图游戏：SG 通用。
- 阶梯博弈（Staircase Nim）：奇偶位异或。

## 2. 三分求极值

适用：单峰 / 单谷函数（凸函数）在区间上求极值（如最优角度、最优比例）。

复杂度：$O(\text{迭代次数})$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

double f(double x)
{
    return -(x - 3) * (x - 3) + 5;
}

double ternary_max(double l, double r)
{
    for (int i = 0; i < 200; i++)
    {
        double m1 = l + (r - l) / 3;
        double m2 = r - (r - l) / 3;
        if (f(m1) < f(m2)) l = m1;
        else r = m2;
    }
    return f((l + r) / 2);
}
```

使用说明：
- 求最大值用上述写法；求最小值把 `f(m1) < f(m2)` 改为 `f(m1) > f(m2)`。
- 迭代 100~200 次即可达双精度收敛；也可用 `while (r - l > 1e-12)`。

变体：
- 整数域三分：`m1 = l + (r-l)/3` 取整，注意收敛条件 `l < r - 2`。
- 函数非严格单峰时结果不可靠，需先验证凸性。

## 3. bitset 优化

适用：0/1 背包可达和、传递闭包、子集关系判断等，常数级加速（÷64）。

复杂度：$O(n^2/64)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

bitset<100005> bs;

void subset_sum(int *a, int n, int maxsum)
{
    bs.reset();
    bs[0] = 1;
    for (int i = 1; i <= n; i++) bs |= bs << a[i];
}

bool reachable(int s)
{
    return bs.test(s);
}
```

使用说明：
- 经典 0/1 背包：选择若干 a[i] 能凑出的所有和，`bs[s]=1` 表示 s 可达。
- 多重背包（数量有限）：按二进制分组后仍用 bitset。
- 传递闭包 / 子集包含：`bs[i] |= bs[j]` 需要多行（按需扩展为 `bitset` 数组）。

变体：
- 求最小凑出数：bitset 无法回溯，改用 int 数组 DP。
- 大 n（>1e6）可用 `vector<bool>` / 手写 64 位块。

## 4. 德州扑克：五张牌判型 + 两副牌比大小（结构体版）

> 本模板按需求保留详细注释（与库内"纯净代码无注释"惯例不同），方便理解牌型判定逻辑；比赛提交时如嫌长可自行删注释。

适用：梭哈 / 德州扑克类比牌问题 —— 给定一副 5 张牌判出牌型，或两副 5 张牌比较大小。

复杂度：判型 $O(1)$，比较 $O(1)$。

约定：
- 五张牌用结构体 `Hand` 存：`r[0..4]` 点数（2..14，14=A），`c[0..4]` 花色（0..3，仅用于判同花），`lv` 牌型等级（9 最高 ~ 1 最低），`key[0..4]` 比大小用的关键点数（降序、多余位补 0）。
- 给五张牌赋初始值：`set_card(h, i, r, c)` 逐张赋值，或 `init_hand(h, r0,c0, r1,c1, ..., r4,c4)` 一次赋 5 张。
- 判型：`eval(h)` 填好 `h.lv` 与 `h.key[]`（比大小前先对每副牌各调一次）。
- 比大小：`cmp(a, b)` 返回 1（a 赢）/ -1（a 输）/ 0（打平），先比牌型 `lv`，相同再逐位比 `key[]`。

牌型等级表（`lv`）：
| lv | 牌型 | 示例 | 比较规则 |
|---|---|---|---|
| 9 | 同花顺（含皇家同花顺） | A♠K♠Q♠J♠T♠ | 比最大点数 |
| 8 | 四条 | 8♠8♥8♦8♣K | 比四条点数，再比单张 |
| 7 | 葫芦 | K♠K♥K♦2♠2♥ | 比三条点数，再比对子 |
| 6 | 同花 | A♠K♠9♠7♠3♠ | 5 张点数依次比 |
| 5 | 顺子 | 9♠8♥7♦6♣5♠ | 比最大点数（A2345 记 5） |
| 4 | 三条 | T♠T♥T♦K♠Q♥ | 比三条点数，再比两张单牌 |
| 3 | 两对 | Q♠Q♥9♠9♥A♦ | 大对、小对、单牌依次比 |
| 2 | 一对 | 7♠7♥A♠K♥Q♦ | 比对子，再比三张单牌 |
| 1 | 高牌 | A♠K♥9♦7♣3♠ | 5 张点数依次比 |

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int RANK = 15;   // 点数取值范围 [2,14]，数组开 15

// 五张牌结构体：r[] 点数(2..14)，c[] 花色(0..3)，lv/key[] 为判型结果
struct Hand
{
    int r[5];    // 5 张牌点数（2..14，14=A）
    int c[5];    // 5 张牌花色（0..3，仅用于判同花）
    int lv;      // 牌型等级（9 最高 ~ 1 最低，0 表示未判型/空）
    int key[5];  // 比大小用关键点数（降序，多余位补 0）
};

// 给第 i 张牌赋初始值（i = 0..4）
void set_card(Hand &h, int i, int r, int c)
{
    h.r[i] = r;
    h.c[i] = c;
}

// 给五张牌赋初始值：依次传 5 张牌的点数与花色（共 10 个参数）
void init_hand(Hand &h, int r0, int c0, int r1, int c1, int r2, int c2, int r3, int c3, int r4, int c4)
{
    h.r[0] = r0; h.c[0] = c0;
    h.r[1] = r1; h.c[1] = c1;
    h.r[2] = r2; h.c[2] = c2;
    h.r[3] = r3; h.c[3] = c3;
    h.r[4] = r4; h.c[4] = c4;
}

// 判定 h 的 5 张牌牌型：结果写入 h.lv（牌型等级）与 h.key[0..4]（比较用关键点数）
void eval(Hand &h)
{
    // 1) 统计各点数出现次数、各花色张数
    int cnt[RANK] = {0}, suit[4] = {0};
    for (int i = 0; i < 5; i++)
    {
        cnt[h.r[i]]++;
        suit[h.c[i]]++;
    }

    // 2) 把 5 张牌按「出现次数降序、点数降序」排成 key[0..4]。
    //    例：四条8888K → [8,8,8,8,K]；葫芦KKK22 → [K,K,K,2,2]；两对QQ99A → [Q,Q,9,9,A]
    //    好处：牌型只看 key 前缀相等关系即可判定，且 key 本身天然就是各牌型的关键点数顺序，
    //          后面直接把它抄进 h.key[] 即可用于比大小。
    int key[5], k = 0;
    for (int c = 5; c >= 1; c--)          // 出现次数从多到少
    {
        for (int r = 14; r >= 2; r--)     // 同次数内点数从大到小
        {
            if (cnt[r] == c)
            {
                for (int t = 0; t < c; t++) key[k++] = r;
            }
        }
    }

    // 3) 判顺子：5 个点数连续。特判 A2345（A 视作 1，为最小顺子）
    bool isStraight = false;
    int straightHigh = 0;                 // 顺子的最大点数（A2345 记 5）
    int mn = 15, mx = 0, distinct = 0;
    for (int r = 2; r <= 14; r++) if (cnt[r]) distinct++;
    if (distinct == 5)                    // 只有 5 个点数互不相同才可能成顺子
    {
        for (int r = 2; r <= 14; r++) if (cnt[r]) { mn = r; break; }
        for (int r = 14; r >= 2; r--) if (cnt[r]) { mx = r; break; }
        if (mx - mn == 4)                 // 正常顺子：max-min == 4
        {
            isStraight = true;
            straightHigh = mx;
        }
        else if (cnt[14] && cnt[2] && cnt[3] && cnt[4] && cnt[5])
        {
            isStraight = true;            // A2345 轮子：A 当 1 用
            straightHigh = 5;
        }
    }

    // 4) 判同花：5 张同花色
    bool isFlush = false;
    for (int s = 0; s < 4; s++) if (suit[s] == 5) isFlush = true;

    // 5) 从大到小依次判定牌型并填 h.key[]。
    //    注：同花/顺子要求 5 个点数互不相同，与对子/三条/葫芦/四条互斥，故分支顺序安全。
    if (isStraight && isFlush)
    {
        h.lv = 9;                         // 同花顺（10JQKA 同花即皇家同花顺，属最高同花顺）
        h.key[0] = straightHigh;
        h.key[1] = h.key[2] = h.key[3] = h.key[4] = 0;
    }
    else if (key[0] == key[3])
    {
        h.lv = 8;                         // 四条：key[0]=key[3] 四个相同，key[4] 为单张
        h.key[0] = key[0];
        h.key[1] = key[4];
        h.key[2] = h.key[3] = h.key[4] = 0;
    }
    else if (key[0] == key[2] && key[3] == key[4])
    {
        h.lv = 7;                         // 葫芦：key 按次数降序排，三条必在前，即 [三,三,三,对,对]
        h.key[0] = key[0];                // 三条点数（葫芦先比三条）
        h.key[1] = key[3];                // 对子点数
        h.key[2] = h.key[3] = h.key[4] = 0;
    }
    else if (isFlush)
    {
        h.lv = 6;                         // 同花：点数从大到小依次比（key 即降序）
        for (int i = 0; i < 5; i++) h.key[i] = key[i];
    }
    else if (isStraight)
    {
        h.lv = 5;                         // 顺子：只比最大点数
        h.key[0] = straightHigh;
        h.key[1] = h.key[2] = h.key[3] = h.key[4] = 0;
    }
    else if (key[0] == key[2])
    {
        h.lv = 4;                         // 三条：key[0]=key[2]，后两位为两张单牌
        h.key[0] = key[0];
        h.key[1] = key[3];
        h.key[2] = key[4];
        h.key[3] = h.key[4] = 0;
    }
    else if (key[0] == key[1] && key[2] == key[3])
    {
        h.lv = 3;                         // 两对：先比大对、再比小对、最后比单牌
        h.key[0] = key[0];
        h.key[1] = key[2];
        h.key[2] = key[4];
        h.key[3] = h.key[4] = 0;
    }
    else if (key[0] == key[1])
    {
        h.lv = 2;                         // 一对：先比对子，再依次比三张单牌
        h.key[0] = key[0];
        h.key[1] = key[2];
        h.key[2] = key[3];
        h.key[3] = key[4];
        h.key[4] = 0;
    }
    else
    {
        h.lv = 1;                         // 高牌：5 张点数降序
        for (int i = 0; i < 5; i++) h.key[i] = key[i];
    }
}

// 两副牌比大小：a 赢 b 返回 1，a 输 b 返回 -1，打平返回 0。
// 先比牌型等级 lv，相同再从左到右逐位比关键点数 key[]（降序）
int cmp(const Hand &a, const Hand &b)
{
    if (a.lv != b.lv) return a.lv > b.lv ? 1 : -1;
    for (int i = 0; i < 5; i++)
    {
        if (a.key[i] != b.key[i]) return a.key[i] > b.key[i] ? 1 : -1;
    }
    return 0;
}
```

使用说明：
- 判一副 5 张牌：`init_hand(h, ...)`（或逐张 `set_card`）赋好牌 → `eval(h)` → 读 `h.lv`（牌型）与 `h.key[]`（关键点数）。
- 两副牌比大小：各自 `eval` 后调 `cmp(a, b)`，返回 1（a 赢）/ -1（a 输）/ 0（打平）。
- 牌型转换：读入字符牌时映射 `2~9→2~9, T→10, J→11, Q→12, K→13, A→14`。
- 花色编号仅用于判同花，0~3 任意映射均可；注意牌面不存在重复（同点数同花色不会同时出现）。
- `cmp` 要求两副牌都已 `eval` 过。

变体：
- 多名玩家排序：把各玩家 Hand 存进数组，用 `cmp` 作为 sort 的比较函数。
- 需要从 7 张牌（2 底牌 + 5 公共牌）选最大 5 张牌型：枚举 C(7,5)=21 种组合，每种填 Hand → eval → 用 cmp 保留最大即可。
- 5 张牌内最大同花判定 / 最接近某牌的顺子：在 eval 里加特殊条件即可。
