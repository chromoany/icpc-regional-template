# 02 · 数据结构

## 1. 树状数组：单点加 + 区间和

适用：下标从 1 开始，需要单点修改 + 区间求和（可推广二维）。

复杂度：修改 / 查询均 $O(\log n)$，空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int n;
ll c[MAXN];

int lowbit(int x)
{
    return x & (-x);
}

void add(int i, ll v)
{
    for (; i <= n; i += lowbit(i)) c[i] += v;
}

ll sum(int i)
{
    ll s = 0;
    for (; i > 0; i -= lowbit(i)) s += c[i];
    return s;
}

ll query(int l, int r)
{
    return sum(r) - sum(l - 1);
}
```

使用说明：
- 初始化：把原数组 a[] 的每个值 `add(i, a[i])`。
- 求逆序对：按值域离散化后从左到右 `query(a[i]+1, n)` 累加。
- 求第 k 小：配合 `lowbit` 倍增跳（见变体）。

变体：
- 区间加 + 单点查询：维护差分数组，`add(l,v), add(r+1,-v)`，单点查 `sum(i)`。
- 区间加 + 区间求和：开两个树状数组（$\text{差分}+\text{差分}\times i$）。
- 求全局第 k 小（权值树状数组）：倍增。

## 2. 线段树：区间加 + 区间求和

适用：任意区间修改（加 / 减）+ 区间查询。维护和的版本。

复杂度：$O(\log n)$ / 次操作，空间 $O(4n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
ll a[MAXN];
ll sum[MAXN << 2];
ll lazy[MAXN << 2];

void push_up(int rt)
{
    sum[rt] = sum[rt << 1] + sum[rt << 1 | 1];
}

void push_down(int rt, int l, int r)
{
    if (lazy[rt])
    {
        int mid = (l + r) >> 1;
        sum[rt << 1] += lazy[rt] * (mid - l + 1);
        sum[rt << 1 | 1] += lazy[rt] * (r - mid);
        lazy[rt << 1] += lazy[rt];
        lazy[rt << 1 | 1] += lazy[rt];
        lazy[rt] = 0;
    }
}

void build(int rt, int l, int r)
{
    lazy[rt] = 0;
    if (l == r)
    {
        sum[rt] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(rt << 1, l, mid);
    build(rt << 1 | 1, mid + 1, r);
    push_up(rt);
}

void update(int rt, int l, int r, int ql, int qr, ll v)
{
    if (ql <= l && r <= qr)
    {
        sum[rt] += v * (r - l + 1);
        lazy[rt] += v;
        return;
    }
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(rt << 1, l, mid, ql, qr, v);
    if (qr > mid) update(rt << 1 | 1, mid + 1, r, ql, qr, v);
    push_up(rt);
}

ll query(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return sum[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query(rt << 1, l, mid, ql, qr);
    if (qr > mid) res += query(rt << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```

使用说明：
- 调用：`build(1,1,n)`；区间加 `update(1,1,n,l,r,v)`；区间和 `query(1,1,n,l,r)`。
- 节点编号 1 开头，`rt<<1|1` 为右子。
- 修改为"赋值"时见模板 3。

变体：
- 维护区间最大值 / 最小值：`push_up` 改取 max/min，`sum` 换 `mx`，lazy 加法不变。
- 区间取反 / 异或：lazy 存 0/1，`sum = len - sum`。
- 动态开点 / 权值线段树：不用 build，`update` 时 `if(!rt) rt=++tot`。

## 3. 线段树：区间赋值 + 区间和 / 区间最值（覆盖型）

适用：操作是"区间整体赋成同一个值"（覆盖）而不是累加，如染色、区间清空重填。

复杂度：$O(\log n)$ / 次操作，空间 $O(4n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const ll NEG = 0x3f3f3f3f3f3f3f3fLL;
ll a[MAXN];
ll sum[MAXN << 2];
ll mx[MAXN << 2];
ll tag[MAXN << 2];

void apply(int rt, int l, int r, ll v)
{
    sum[rt] = v * (r - l + 1);
    mx[rt] = v;
    tag[rt] = v;
}

void push_down(int rt, int l, int r)
{
    if (tag[rt] != NEG)
    {
        int mid = (l + r) >> 1;
        apply(rt << 1, l, mid, tag[rt]);
        apply(rt << 1 | 1, mid + 1, r, tag[rt]);
        tag[rt] = NEG;
    }
}

void push_up(int rt)
{
    sum[rt] = sum[rt << 1] + sum[rt << 1 | 1];
    mx[rt] = max(mx[rt << 1], mx[rt << 1 | 1]);
}

void build(int rt, int l, int r)
{
    tag[rt] = NEG;
    if (l == r)
    {
        sum[rt] = mx[rt] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(rt << 1, l, mid);
    build(rt << 1 | 1, mid + 1, r);
    push_up(rt);
}

void assign(int rt, int l, int r, int ql, int qr, ll v)
{
    if (ql <= l && r <= qr)
    {
        apply(rt, l, r, v);
        return;
    }
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) assign(rt << 1, l, mid, ql, qr, v);
    if (qr > mid) assign(rt << 1 | 1, mid + 1, r, ql, qr, v);
    push_up(rt);
}

ll query_sum(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return sum[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query_sum(rt << 1, l, mid, ql, qr);
    if (qr > mid) res += query_sum(rt << 1 | 1, mid + 1, r, ql, qr);
    return res;
}

ll query_max(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return mx[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = NEG;
    if (ql <= mid) res = max(res, query_max(rt << 1, l, mid, ql, qr));
    if (qr > mid) res = max(res, query_max(rt << 1 | 1, mid + 1, r, ql, qr));
    return res;
}
```

使用说明：
- 与模板 2 的区别：lazy 换成 `tag`，标记"整个区间被覆盖为 v"，`NEG` 表示无标记（注意别让真实值取到 NEG）。
- 先 build 再 assign / query。
- 若还要支持"区间加"和"区间赋值"同时存在，需两个懒标记并规定优先级（赋值覆盖加法）。

变体：
- 维护 gcd：`push_up` 取 gcd，区间加（差分）可做区间 gcd。
- 与线段树合并 / 扫描线结合做矩形面积并 / 周长。

## 4. 并查集：路径压缩 + 按秩合并

适用：动态连通性、最小生成树 Kruskal、离线处理区间关系。

复杂度：均摊 $O(\alpha(n))$（反阿克曼，近似常数），空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int fa[MAXN];
int rk[MAXN];

int find(int x)
{
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

void merge(int x, int y)
{
    x = find(x);
    y = find(y);
    if (x == y) return;
    if (rk[x] < rk[y]) swap(x, y);
    fa[y] = x;
    if (rk[x] == rk[y]) rk[x]++;
}

void init(int n)
{
    for (int i = 1; i <= n; i++)
    {
        fa[i] = i;
        rk[i] = 0;
    }
}
```

使用说明：
- 先 `init(n)`。`find` 带路径压缩（迭代版更快：`while(fa[x]!=x) x=fa[x]=fa[fa[x]]`）。
- 按秩合并保证树高 $O(\log n)$，配合路径压缩均摊常数。

变体：
- 带权并查集：维护到根的权值（食物链、种类），`find` 时累加权值。
- 可撤销并查集：用栈记录合并操作，按秩合并后支持回滚（离线分治）。

## 5. ST 表：静态区间最值 RMQ

适用：数组只读不改，大量区间 max/min 查询。不能用于求和（和没有幂等性）。

复杂度：预处理 $O(n\log n)$，查询 $O(1)$，空间 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int LOG = 18;
int st[MAXN][LOG];
int lg2[MAXN];

void init_st(int n)
{
    for (int i = 2; i <= n; i++) lg2[i] = lg2[i >> 1] + 1;
    for (int j = 1; (1 << j) <= n; j++) for (int i = 1; i + (1 << j) - 1 <= n; i++) st[i][j] = max(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
}

int query_max(int l, int r)
{
    int k = lg2[r - l + 1];
    return max(st[l][k], st[r - (1 << k) + 1][k]);
}
```

使用说明：
- 先把原数组放入 `st[i][0]`，再调用 `init_st(n)`。
- LOG 取满足 $2^{\text{LOG}}\ge n$ 的最小值（n=1e5 用 17，留余量取 18）。

变体：
- 区间 gcd：`max` 换 `__gcd`，仍幂等可查。
- 区间 min：`max` 换 `min`。
- 树上 RMQ：欧拉序 + ST 表求 LCA（可代替倍增）。

## 6. 主席树：静态区间第 k 小

适用：静态数组，询问 [l,r] 内第 k 小（或 k 大、区间不同数个数等）。

复杂度：建树 $O(n\log n)$，单次查询 $O(\log n)$，空间 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int ls[MAXN * 20];
int rs[MAXN * 20];
int cnt[MAXN * 20];
int root[MAXN];
int tot;

int build(int l, int r)
{
    int rt = ++tot;
    cnt[rt] = 0;
    if (l < r)
    {
        int mid = (l + r) >> 1;
        ls[rt] = build(l, mid);
        rs[rt] = build(mid + 1, r);
    }
    return rt;
}

int update(int pre, int l, int r, int pos)
{
    int rt = ++tot;
    ls[rt] = ls[pre];
    rs[rt] = rs[pre];
    cnt[rt] = cnt[pre] + 1;
    if (l < r)
    {
        int mid = (l + r) >> 1;
        if (pos <= mid) ls[rt] = update(ls[pre], l, mid, pos);
        else rs[rt] = update(rs[pre], mid + 1, r, pos);
    }
    return rt;
}

int query_kth(int u, int v, int l, int r, int k)
{
    if (l == r) return l;
    int mid = (l + r) >> 1;
    int left = cnt[ls[v]] - cnt[ls[u]];
    if (k <= left) return query_kth(ls[u], ls[v], l, mid, k);
    return query_kth(rs[u], rs[v], mid + 1, r, k - left);
}
```

使用说明：
- 值域离散化到 [1, m]，原数组 a[1..n]。
- `root[0] = build(1, m)`；对 i=1..n：`root[i] = update(root[i-1], 1, m, id(a[i]))`。
- 询问 [l,r] 第 k 小：`query_kth(root[l-1], root[r], 1, m, k)`，返回离散化下标，映射回原值。

变体：
- 求区间小于 x 的个数：`query_less` 统计前缀和小于某值的数量。
- 可持久化权值线段树可扩展到树上路径第 k 小（配合树剖 / dfs 序）。

## 7. 莫队：区间不同颜色数（离线）

适用：只有区间询问、无修改，维护的贡献可 $O(1)$ 增减（不同数、众数、出现次数奇偶等）。

复杂度：$O((n+q)\sqrt n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int a[MAXN];
int cnt[MAXN];
ll ans[MAXN];
int ql[MAXN], qr[MAXN], qid[MAXN], blk[MAXN];
int ord[MAXN];
int n, m, q;
ll cur_ans;

bool cmp(int x, int y)
{
    if (blk[ql[x]] != blk[ql[y]]) return blk[ql[x]] < blk[ql[y]];
    return qr[x] < qr[y];
}

void add(int p)
{
    cnt[a[p]]++;
    if (cnt[a[p]] == 1) cur_ans++;
}

void del(int p)
{
    cnt[a[p]]--;
    if (cnt[a[p]] == 0) cur_ans--;
}

void solve()
{
    int size = max(1, (int)sqrt(n));
    for (int i = 1; i <= n; i++) blk[i] = (i - 1) / size + 1;
    for (int i = 1; i <= q; i++) ord[i] = i;
    sort(ord + 1, ord + q + 1, cmp);
    int L = 1, R = 0;
    cur_ans = 0;
    for (int i = 1; i <= q; i++)
    {
        int idx = ord[i];
        while (L > ql[idx])
        {
            add(--L);
        }
        while (R < qr[idx])
        {
            add(++R);
        }
        while (L < ql[idx])
        {
            del(L++);
        }
        while (R > qr[idx])
        {
            del(R--);
        }
        ans[qid[idx]] = cur_ans;
    }
}
```

使用说明：
- 询问存于 `ql/qr/qid`（下标 1..q），a[1..n] 为原数组。
- 调用 `solve()` 后答案写入 `ans[]`，按原询问顺序输出。
- 奇偶优化：`cmp` 里 R 的方向按块奇偶反转，常数减半（模板已可加）。

变体：
- 带修改莫队：增加时间维 $t$，块大小取 $n^{2/3}$，复杂度 $O(n^{5/3})$。
- 树上莫队：欧拉序转化为序列莫队。

## 8. 分块：区间加 + 区间和

适用：操作复杂（区间加、区间和、区间排序、区间众数等）线段树难以维护时，根号分治兜底。

复杂度：$O((n+q)\sqrt n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int BS = 320;
ll a[MAXN];
ll sum[MAXN / BS + 5];
ll lazy[MAXN / BS + 5];
int L[MAXN / BS + 5], R[MAXN / BS + 5], bel[MAXN];
int n, q, bn;

void build()
{
    bn = (n + BS - 1) / BS;
    for (int i = 1; i <= bn; i++)
    {
        L[i] = (i - 1) * BS + 1;
        R[i] = min(i * BS, n);
        sum[i] = 0;
        for (int j = L[i]; j <= R[i]; j++)
        {
            bel[j] = i;
            sum[i] += a[j];
        }
    }
}

void update(int l, int r, ll v)
{
    int bl = bel[l], br = bel[r];
    if (bl == br)
    {
        for (int i = l; i <= r; i++)
        {
            a[i] += v;
            sum[bl] += v;
        }
        return;
    }
    for (int i = l; i <= R[bl]; i++)
    {
        a[i] += v;
        sum[bl] += v;
    }
    for (int i = bl + 1; i <= br - 1; i++) lazy[i] += v;
    for (int i = L[br]; i <= r; i++)
    {
        a[i] += v;
        sum[br] += v;
    }
}

ll query(int l, int r)
{
    int bl = bel[l], br = bel[r];
    ll res = 0;
    if (bl == br)
    {
        for (int i = l; i <= r; i++) res += a[i] + lazy[bl];
        return res;
    }
    for (int i = l; i <= R[bl]; i++) res += a[i] + lazy[bl];
    for (int i = bl + 1; i <= br - 1; i++) res += sum[i] + lazy[i] * (R[i] - L[i] + 1);
    for (int i = L[br]; i <= r; i++) res += a[i] + lazy[br];
    return res;
}
```

使用说明：
- 先 `build()` 再操作。`update(l,r,v)` 区间加，`query(l,r)` 区间和。
- 整块打 lazy，零散块暴力改并更新 sum。
- 块大小 BS 按 $n$ 调整，$\sqrt n$ 附近最优。

变体：
- 区间排序：块内维护有序 vector，查询二分，修改整块重建。
- 区间众数 / 出现次数：预处理块间答案 + 两侧暴力。

## 9. FHQ-Treap（无旋平衡树：分裂 / 合并）

适用：需要有序集合全套操作（插入 / 删除 / 排名 / 第 k 小 / 前驱后继），或区间操作（区间翻转、区间加）；比 Splay 短且好写，支持可持久化。

复杂度：单次操作期望 $O(\log n)$。

约定：`pr` 为大根堆随机优先级；`split(u, v, x, y)` 按值把树 $u$ 分成 $x$（$\le v$）与 $y$（$> v$）；`merge(x, y)` 要求 $x$ 中所有值 $\le$ $y$ 中所有值。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 200005;
int lc[MAXN], rc[MAXN], sz[MAXN], pr[MAXN];
ll val[MAXN];
int root, tot;

int newnode(ll v)
{
    int u = ++tot;
    val[u] = v;
    sz[u] = 1;
    pr[u] = rand();
    lc[u] = rc[u] = 0;
    return u;
}

void push_up(int u)
{
    sz[u] = sz[lc[u]] + sz[rc[u]] + 1;
}

void split(int u, ll v, int &x, int &y)
{
    if (!u)
    {
        x = y = 0;
        return;
    }
    if (val[u] <= v)
    {
        x = u;
        split(rc[u], v, rc[u], y);
    }
    else
    {
        y = u;
        split(lc[u], v, x, lc[u]);
    }
    push_up(u);
}

int merge(int x, int y)
{
    if (!x || !y) return x + y;
    if (pr[x] < pr[y])
    {
        rc[x] = merge(rc[x], y);
        push_up(x);
        return x;
    }
    lc[y] = merge(x, lc[y]);
    push_up(y);
    return y;
}

void insert_val(ll v)
{
    int a, b;
    split(root, v, a, b);
    root = merge(merge(a, newnode(v)), b);
}

void erase_val(ll v)
{
    int a, b, c;
    split(root, v, a, b);
    split(a, v - 1, a, c);
    c = merge(lc[c], rc[c]);
    root = merge(merge(a, c), b);
}

int kth(int u, int k)
{
    while (1)
    {
        if (sz[lc[u]] >= k) u = lc[u];
        else if (sz[lc[u]] + 1 == k) return u;
        else
        {
            k -= sz[lc[u]] + 1;
            u = rc[u];
        }
    }
}

int rank_of(ll v)
{
    int a, b, res;
    split(root, v - 1, a, b);
    res = sz[a] + 1;
    root = merge(a, b);
    return res;
}

ll prev_val(ll v)
{
    int a, b, u;
    split(root, v - 1, a, b);
    if (!a)
    {
        root = merge(a, b);
        return LLONG_MIN;
    }
    u = kth(a, sz[a]);
    root = merge(a, b);
    return val[u];
}

ll next_val(ll v)
{
    int a, b, u;
    split(root, v, a, b);
    if (!b)
    {
        root = merge(a, b);
        return LLONG_MAX;
    }
    u = kth(b, 1);
    root = merge(a, b);
    return val[u];
}
```

使用说明：
- 初始化：`root = 0; tot = 0;`（不要忘记把 `lc[0] = rc[0] = sz[0] = 0`）。
- 按值分裂适合"有序集合"；按大小分裂（`split_size(u, k, x, y)`，判据换成 `sz[lc[u]] + 1 <= k`）适合"区间操作 / 序列维护"。
- 随机优先级用 `rand()` 够用；被卡时换 `mt19937_64`。

变体：
- 区间翻转：节点加 `rev` 标记，`push_down` 里交换左右儿子并下传，`split_size` 前必须一路 `push_down`。
- 可持久化：`newnode` 改成"复制当前节点再改"（`int clone(int u)`），`split`/`merge` 返回新节点即可得到版本树。
- 区间加 + 区间和：再加 `sum` 与 `lazy`，`push_up` 与 `push_down` 同步维护。

## 10. 李超线段树（区间最值线段）

适用：动态维护若干直线 / 线段，询问某个 $x$ 处的最大值（或最小值）；斜率优化 DP 的"斜率不单调 / 横坐标任意"情形。

复杂度：插入 $O(\log^2 n)$（线段）或 $O(\log n)$（直线），查询 $O(\log n)$。

约定：定义域离散为 $x \in [1, n]$（横坐标先离散化）；`empty` 表示该节点尚无直线。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const double INF = 1e18;

struct Line
{
    double k, b;
    bool empty;
} tree[MAXN << 2];

double calc(Line li, double x)
{
    return li.k * x + li.b;
}

void insert_line(int rt, int l, int r, Line li)
{
    if (tree[rt].empty)
    {
        tree[rt] = li;
        return;
    }
    int mid = (l + r) >> 1;
    if (calc(li, mid) > calc(tree[rt], mid)) swap(li, tree[rt]);
    if (l == r) return;
    if (calc(li, l) > calc(tree[rt], l)) insert_line(rt << 1, l, mid, li);
    else if (calc(li, r) > calc(tree[rt], r)) insert_line(rt << 1 | 1, mid + 1, r, li);
}

double query_max(int rt, int l, int r, int x)
{
    double res = tree[rt].empty ? -INF : calc(tree[rt], (double)x);
    if (l == r) return res;
    int mid = (l + r) >> 1;
    if (x <= mid) return max(res, query_max(rt << 1, l, mid, x));
    return max(res, query_max(rt << 1 | 1, mid + 1, r, x));
}
```

使用说明：
- 定义域是**离散**的 $[1,n]$，把题目给的横坐标离散化后按编号查；查询传编号（或用实际 $x$ 值，只要插入时也用同一套坐标）。
- 初始化：把所有节点 `tree[i].empty = true`。
- 求最小值：把 `>` 全改成 `<`，初值改 `INF`；或插入 $-k, -b$ 后取负。

变体：
- 插入"线段"（有 $x$ 范围限制）：在区间定位的 $O(\log n)$ 个节点上各插一次，复杂度 $O(\log^2 n)$。
- 求区间最值（不单点）：节点维护区间最大值，查询时带上路径上的 $\max$。
- 直线只有插入没有修改且询问离线时，可直接用斜率优化 / 凸包代替。

## 11. CDQ 分治（三维偏序）

适用：三维偏序计数（$a_i \le a_j, b_i \le b_j, c_i \le c_j$ 的点对数）、带时间的动态问题转静态（删去时间维）、斜率优化 DP 的左半更新右半。

复杂度：$O(n\log^2 n)$。

约定：先按 $a$ 排序并把相同三元组合并（`cnt` 记重复个数），再跑 `cdq(1, cnt_unique)`；`ans[i]` 为偏序数 $\ge i$ 的个数累计。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
struct Node
{
    int a, b, c, cnt, ans;
} p[MAXN], tmp[MAXN];
int bit[MAXN], res[MAXN];
int n, k;

int lowbit(int x)
{
    return x & (-x);
}

void add(int i, int v)
{
    for (; i <= k; i += lowbit(i)) bit[i] += v;
}

int sum(int i)
{
    int s = 0;
    for (; i > 0; i -= lowbit(i)) s += bit[i];
    return s;
}

void cdq(int l, int r)
{
    if (l == r) return;
    int mid = (l + r) >> 1;
    cdq(l, mid);
    cdq(mid + 1, r);
    int i = l, j = mid + 1, t = 0;
    while (i <= mid && j <= r)
    {
        if (p[i].b <= p[j].b)
        {
            add(p[i].c, p[i].cnt);
            tmp[t++] = p[i++];
        }
        else
        {
            p[j].ans += sum(p[j].c);
            tmp[t++] = p[j++];
        }
    }
    while (i <= mid) tmp[t++] = p[i++];
    while (j <= r)
    {
        p[j].ans += sum(p[j].c);
        tmp[t++] = p[j++];
    }
    for (int x = l; x <= mid; x++) add(p[x].c, -p[x].cnt);
    for (int x = 0; x < t; x++) p[l + x] = tmp[x];
}
```

使用说明：
- 流程：读入 $n$ 个三元组 → `sort`（按 $a,b,c$ 字典序）→ 相邻去重合并成 $m$ 个节点并记 `cnt` → `cdq(1, m)` → 对每个节点 `res[p[i].ans + p[i].cnt - 1] += p[i].cnt`，其中 `res[t]` 表示"被恰好 $t$ 个点偏序覆盖的点数"。
- 树状数组值域是 $c$ 的范围，用完后必须**清空**（代码里 `add(..., -p[x].cnt)`），否则污染。
- `p[i].ans` 只统计"左半对右半"的贡献，归并顺序保证 $b$ 有序。

变体：
- 四维偏序：外层再套一层 CDQ（$O(n\log^3 n)$）。
- 斜率优化 DP：把 $j$ 转移写成 $f_i = \min_j (k_i x_j + y_j)$ 的形式，用 CDQ 按 $k$ 排序 + 单调凸包维护，处理 $k$、$x$ 都不单调的情形。
- 单点修改 + 区间查询离线：把操作序列当时间维，CDQ 处理"修改对后续查询"的贡献。

## 12. 整体二分（区间第 k 小 / 带修改区间第 k 小）

适用：多组询问的"二分答案"问题（区间第 k 小、区间第 k 大、区间内 $\le x$ 的个数），且 check 可离线 —— 一次分治同时处理所有询问。

复杂度：$O((n+q)\log V \log n)$。

约定：取值范围离散化到 $[1, V]$，`head[v]` 存"值 $v$ 在序列中的所有位置"的链表；询问 `q[i]` 含 `l, r, k, id`。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int a[MAXN], ans[MAXN];
int head[MAXN], nxp[MAXN];
struct Query
{
    int l, r, k, id;
} q[MAXN], q1[MAXN], q2[MAXN];
int bit[MAXN];
int n, m, V;

int lowbit(int x)
{
    return x & (-x);
}

void add(int i, int v)
{
    for (; i <= n; i += lowbit(i)) bit[i] += v;
}

int sum(int i)
{
    int s = 0;
    for (; i > 0; i -= lowbit(i)) s += bit[i];
    return s;
}

void add_pos(int v, int pos)
{
    nxp[pos] = head[v];
    head[v] = pos;
}

void solve(int l, int r, int ql, int qr)
{
    if (ql > qr) return;
    if (l == r)
    {
        for (int i = ql; i <= qr; i++) ans[q[i].id] = l;
        return;
    }
    int mid = (l + r) >> 1, c1 = 0, c2 = 0;
    for (int v = l; v <= mid; v++) for (int u = head[v]; u; u = nxp[u]) add(u, 1);
    for (int i = ql; i <= qr; i++)
    {
        int t = sum(q[i].r) - sum(q[i].l - 1);
        if (t >= q[i].k) q1[++c1] = q[i];
        else
        {
            q[i].k -= t;
            q2[++c2] = q[i];
        }
    }
    for (int v = l; v <= mid; v++) for (int u = head[v]; u; u = nxp[u]) add(u, -1);
    for (int i = 1; i <= c1; i++) q[ql + i - 1] = q1[i];
    for (int i = 1; i <= c2; i++) q[ql + c1 + i - 1] = q2[i];
    solve(l, mid, ql, ql + c1 - 1);
    solve(mid + 1, r, ql + c1, qr);
}
```

使用说明：
- 先把原数组的值离散化到 $[1,V]$，对每个 $i$ 调 `add_pos(a[i], i)`（同值多个位置会串成链表）。
- 初始调用 `solve(1, V, 1, m)`，答案在 `ans[id]`（离散化后的值）。
- 值域和询问规模都在 $10^5$ 量级时最快；询问数量很少时直接用主席树更省事。

变体：
- 带修改：把修改也当"事件"放进分治（修改 = 删旧值 + 插新值），整体复杂度加一维时间。
- 求区间第 k 大：把 `t >= k` 的方向反过来，或对值取反。
- check 是"区间内满足某条件的元素个数"时，把树状数组换成对应的数据结构（如 BIT 套值域）。

## 13. 线段树合并（动态开点 + 权值合并）

适用：每个点/子树维护一棵权值线段树，自底向上合并统计（树上颜色数、子树内出现次数、出现次数最多的值），配合 DSU on tree 或直接合并。

复杂度：总复杂度 $O(n\log V)$（均摊，每次合并销毁一个节点）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int lc[MAXN * 40], rc[MAXN * 40], sum[MAXN * 40];
int root[MAXN];
int tot, n, V;

void update(int &rt, int l, int r, int pos, int v)
{
    if (!rt) rt = ++tot;
    sum[rt] += v;
    if (l == r) return;
    int mid = (l + r) >> 1;
    if (pos <= mid) update(lc[rt], l, mid, pos, v);
    else update(rc[rt], mid + 1, r, pos, v);
}

int merge_tree(int x, int y, int l, int r)
{
    if (!x || !y) return x + y;
    if (l == r)
    {
        sum[x] += sum[y];
        return x;
    }
    int mid = (l + r) >> 1;
    lc[x] = merge_tree(lc[x], lc[y], l, mid);
    rc[x] = merge_tree(rc[x], rc[y], mid + 1, r);
    sum[x] = sum[lc[x]] + sum[rc[x]];
    return x;
}

int query_kth(int rt, int l, int r, int k)
{
    if (l == r) return l;
    int mid = (l + r) >> 1;
    if (sum[lc[rt]] >= k) return query_kth(lc[rt], l, mid, k);
    return query_kth(rc[rt], mid + 1, r, k - sum[lc[rt]]);
}
```

使用说明：
- `root[u]` 初始为 0；对点 $u$ 的每个值调 `update(root[u], 1, V, pos, v)`。
- 树形结构上自底向上 `root[u] = merge_tree(root[u], root[v], 1, V)`（$v$ 是 $u$ 的儿子），合并后儿子那棵树不必再访问。
- 空间按 $"$操作次数$" \times \log V$ 估：$n=10^5$、$\log V = 17$ 时约 $2 \times 10^6$ 个节点。

变体：
- 合并时维护额外信息（最大值、最大值位置、异或和）：在 `merge_tree` 的 `l == r` 分支里同步维护。
- 可持久化合并：`merge` 时复制节点（`++tot` 再赋值），可查历史版本。
- 统计"子树内出现次数 $\ge k$ 的颜色数"：节点存 $cnt$ 与 $ok$ 计数，`push_up` 时把儿子的 $ok$ 相加上来。

## 14. 可并堆（左偏树）

适用：需要反复合并两个堆并取最值（可并堆），如"每次合并两个集合、取最小值"、树上贪心合并、斜率优化中的凸包并。

复杂度：合并 / 弹出 $O(\log n)$，取最值 $O(1)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int lc[MAXN], rc[MAXN], dis[MAXN], rt_of[MAXN];
ll val[MAXN];
bool dead[MAXN];

void init_heap(int n)
{
    for (int i = 1; i <= n; i++)
    {
        rt_of[i] = i;
        lc[i] = rc[i] = 0;
        dis[i] = 1;
        dead[i] = false;
    }
    dis[0] = 0;
}

int merge_heap(int x, int y)
{
    if (!x || !y) return x + y;
    if (val[x] > val[y] || (val[x] == val[y] && x > y)) swap(x, y);
    rc[x] = merge_heap(rc[x], y);
    if (dis[lc[x]] < dis[rc[x]]) swap(lc[x], rc[x]);
    dis[x] = dis[rc[x]] + 1;
    return x;
}

int find_rt(int x)
{
    return rt_of[x] == x ? x : rt_of[x] = find_rt(rt_of[x]);
}

void union_heap(int x, int y)
{
    x = find_rt(x);
    y = find_rt(y);
    if (x == y) return;
    int r = merge_heap(x, y);
    rt_of[x] = rt_of[y] = r;
}

ll top_heap(int x)
{
    return val[find_rt(x)];
}

void pop_heap(int x)
{
    x = find_rt(x);
    dead[x] = true;
    rt_of[x] = merge_heap(lc[x], rc[x]);
    rt_of[rt_of[x]] = rt_of[x];
}
```

使用说明：
- 先 `init_heap(n)`；`val[i]` 为第 $i$ 个元素的值。
- `union_heap(x, y)` 合并两元素所在堆，`top_heap(x)` 取堆顶值，`pop_heap(x)` 弹出堆顶（堆顶元素 `dead` 置真，用 `find_rt` 可用性判定）。
- 小根堆；要大根堆把 `val` 取负，或把 `>` 改成 `<`。

变体：
- 斜堆：去掉 `dis` 与左右交换判断，均摊仍是 $O(\log n)$（更好写，常数略大）。
- 配对堆 / 斐波那契堆：需要 decrease-key 时用。
- 左偏树做"树上每个点合并儿子堆并弹掉不合法元素"是经典贪心套路（如任务调度、树上最小代价合并）。

## 15. 笛卡尔树

适用：把序列建成"下标满足 BST、值满足堆"的树，处理区间最值与"以每个元素为最值的区间"（直方图最大矩形、区间最小值划分）。

复杂度：$O(n)$。

约定：`build_cartesian(n)` 建**小根堆**笛卡尔树，返回根结点；`ls/rs` 为左右儿子。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int a[MAXN], ls[MAXN], rs[MAXN], stk[MAXN];

int build_cartesian(int n)
{
    int top = 0, root = 0;
    for (int i = 1; i <= n; i++)
    {
        ls[i] = rs[i] = 0;
        int last = 0;
        while (top && a[stk[top]] > a[i])
        {
            last = stk[top--];
        }
        ls[i] = last;
        if (top) rs[stk[top]] = i;
        else root = i;
        stk[++top] = i;
    }
    return root;
}
```

使用说明：
- 小根堆：`while (top && a[stk[top]] > a[i])`；大根堆把 `>` 改成 `<`。
- 建树后中序遍历即原序列，子树就是一段连续区间，根为该区间最小值 —— 许多区间问题可转成树上问题。
- 值可以相同，此时中序遍历仍是原序列，但注意相等值之间的父子关系由扫描顺序决定。

变体：
- 直方图最大矩形 / 最大子矩阵：对每个 $i$，以其为最小值的最长区间 $=$ 左右子树大小 + 1。
- $O(n)$ 建"区间最小值划分"树后做树形 DP（合法划分方案数、最小代价）。
- 与单调栈互为表里：笛卡尔树是单调栈过程的树化记录。

## 16. 树上启发式合并（DSU on tree）

适用：静态树上"对每个子树回答某种统计"（子树内出现次数最多的颜色、子树内某种点的个数、子树集合信息），离线、无修改。

复杂度：$O(n \log n)$。

约定：以示例"子树内出现次数最多的颜色的**编号和**"为统计目标；`c[u]` 为颜色，改 `proc` 里的两行即可换统计。

```cpp
#include <bits/stdc++.h>
using namespace std;

const int MAXN = 200005;
int n, hd[MAXN], nx[MAXN * 2], to[MAXN * 2], ecnt = 1;
int sz[MAXN], son[MAXN], c[MAXN], cnt[MAXN], ans[MAXN];
int mx, sum;

void add_edge(int u, int v)
{
    to[ecnt] = v;
    nx[ecnt] = hd[u];
    hd[u] = ecnt++;
}

void dfs0(int u, int f)
{
    sz[u] = 1;
    son[u] = 0;
    for (int e = hd[u]; e; e = nx[e])
    {
        int v = to[e];
        if (v == f) continue;
        dfs0(v, u);
        sz[u] += sz[v];
        if (sz[v] > sz[son[u]]) son[u] = v;
    }
}

void proc(int u, int f, int skip, int k)
{
    cnt[c[u]] += k;
    if (k > 0)
    {
        if (cnt[c[u]] > mx) mx = cnt[c[u]], sum = c[u];
        else if (cnt[c[u]] == mx) sum += c[u];
    }
    for (int e = hd[u]; e; e = nx[e])
    {
        int v = to[e];
        if (v != f && v != skip) proc(v, u, 0, k);
    }
}

void dfs(int u, int f, int keep)
{
    for (int e = hd[u]; e; e = nx[e])
    {
        int v = to[e];
        if (v != f && v != son[u])
        {
            dfs(v, u, 0);
            proc(v, u, 0, -1);
        }
    }
    if (son[u]) dfs(son[u], u, 1);
    mx = 0;
    sum = 0;
    proc(u, f, son[u], 1);
    ans[u] = sum;
    if (!keep) proc(u, f, 0, -1);
}
```

使用说明：
- 主流程：`dfs0(rt, 0); dfs(rt, 0, 1);`，答案在 `ans[u]`。
- 统计逻辑全部写在 `proc` 的 `k > 0` 分支（加贡献）里；`k = -1` 分支只回滚计数数组，不动统计变量。
- `mx / sum` 在每个结点统计前重置，重儿子子树的 `cnt` 保留贡献。

变体：
- 统计"子树内大于 x 的个数"等可把 `cnt` 换成树状数组，复杂度 $O(n \log^2 n)$。
- 需要在线 / 可持久化用主席树（02 第 6 节）；子树信息可合并时优先考虑线段树合并（02 第 13 节）。
- 与长链剖分：统计与"深度"相关的量（子树内深度为 d 的点数）时用长链剖分 $O(n)$。

## 17. 珂朵莉树（ODT）

适用：**带区间赋值操作**的区间问题，且数据随机或保证赋值操作足够多（均摊 $O(n \log n)$）。经典：随机数据下的区间赋值 + 区间第 k 小 / 幂次和。

复杂度：均摊 $O(n \log n)$（依赖随机数据）。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

struct Node
{
    int l, r;
    mutable ll v;
};
bool operator<(const Node &a, const Node &b)
{
    return a.l < b.l;
}
set<Node> s;
typedef set<Node>::iterator IT;

IT split(int pos)
{
    IT it = s.lower_bound(Node{pos, 0, 0});
    if (it != s.end() && it->l == pos) return it;
    it--;
    int l = it->l, r = it->r;
    ll v = it->v;
    s.erase(it);
    s.insert(Node{l, pos - 1, v});
    return s.insert(Node{pos, r, v}).first;
}

void assign_val(int l, int r, ll v)
{
    IT itr = split(r + 1), itl = split(l);
    s.erase(itl, itr);
    s.insert(Node{l, r, v});
}

void add_val(int l, int r, ll v)
{
    IT itr = split(r + 1), itl = split(l);
    for (IT it = itl; it != itr; it++) it->v += v;
}

ll kth(int l, int r, int k)
{
    IT itr = split(r + 1), itl = split(l);
    vector<pair<ll, int>> tmp;
    for (IT it = itl; it != itr; it++) tmp.push_back(make_pair(it->v, it->r - it->l + 1));
    sort(tmp.begin(), tmp.end());
    for (int i = 0; i < (int)tmp.size(); i++)
    {
        k -= tmp[i].second;
        if (k <= 0) return tmp[i].first;
    }
    return -1;
}

ll qpow(ll a, ll b, ll p)
{
    ll res = 1;
    a %= p;
    while (b)
    {
        if (b & 1) res = res * a % p;
        a = a * a % p;
        b >>= 1;
    }
    return res;
}

ll pow_sum(int l, int r, ll x, ll p)
{
    IT itr = split(r + 1), itl = split(l);
    ll res = 0;
    for (IT it = itl; it != itr; it++)
        res = (res + (ll)(it->r - it->l + 1) % p * qpow(it->v, x, p)) % p;
    return res;
}
```

使用说明：
- 初始化：`s.insert(Node{1, n, 初始值})`，下标 1..n。
- `split(r + 1)` 必须写在 `split(l)` **之前**（否则区间端点被破坏）。
- 均摊复杂度的来源是 `assign_val`；没有区间赋值操作时不要用 ODT，老实用线段树。
- `kth` 里的 `tmp` 是临时排序用的辅助 vector，不属于码风限制的"算法数据存储"。

变体：
- 区间赋值成 f(v)（如 v 的位数）：遍历 `itl..itr` 逐段改 `it->v` 即可。
- 珂朵莉树 + 均摊势能分析（区间取 min 的吉司机线段树场景）：随机数据下也能混用。

## 18. Link-Cut Tree（LCT）

适用：动态树的连通性 / 路径信息维护：加边、删边、换根、路径查询与修改（配合懒标记）。不要求树形态固定。

复杂度：均摊 $O(\log n)$。

约定：示例维护**路径点权和**（`val` 为点权，`sum` 为子树和）；加懒标记时在 `pushdown` 里下传。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int fa[MAXN], ch[MAXN][2], rev[MAXN], val[MAXN], stk[MAXN];
ll sum[MAXN];

bool isroot(int x)
{
    return ch[fa[x]][0] != x && ch[fa[x]][1] != x;
}

void pushup(int x)
{
    sum[x] = sum[ch[x][0]] + sum[ch[x][1]] + val[x];
}

void pushrev(int x)
{
    swap(ch[x][0], ch[x][1]);
    rev[x] ^= 1;
}

void pushdown(int x)
{
    if (rev[x])
    {
        if (ch[x][0]) pushrev(ch[x][0]);
        if (ch[x][1]) pushrev(ch[x][1]);
        rev[x] = 0;
    }
}

void rotate(int x)
{
    int y = fa[x], z = fa[y], k = (ch[y][1] == x), w = ch[x][k ^ 1];
    if (!isroot(y)) ch[z][ch[z][1] == y] = x;
    ch[x][k ^ 1] = y;
    ch[y][k] = w;
    fa[y] = x;
    fa[x] = z;
    if (w) fa[w] = y;
    pushup(y);
    pushup(x);
}

void splay(int x)
{
    int top = 0, y = x;
    stk[++top] = y;
    while (!isroot(y)) stk[++top] = y = fa[y];
    while (top) pushdown(stk[top--]);
    while (!isroot(x))
    {
        int y = fa[x], z = fa[y];
        if (!isroot(y) && (ch[y][1] == x) == (ch[z][1] == y)) rotate(y);
        rotate(x);
    }
}

void access(int x)
{
    for (int y = 0; x; y = x, x = fa[x])
    {
        splay(x);
        ch[x][1] = y;
        pushup(x);
    }
}

void makeroot(int x)
{
    access(x);
    splay(x);
    pushrev(x);
}

int findroot(int x)
{
    access(x);
    splay(x);
    pushdown(x);
    while (ch[x][0])
    {
        x = ch[x][0];
        pushdown(x);
    }
    splay(x);
    return x;
}

void split(int x, int y)
{
    makeroot(x);
    access(y);
    splay(y);
}

void link(int x, int y)
{
    makeroot(x);
    if (findroot(y) != x) fa[x] = y;
}

void cut(int x, int y)
{
    makeroot(x);
    if (findroot(y) != x || fa[y] != x || ch[y][0]) return;
    fa[y] = 0;
    ch[x][1] = 0;
    pushup(x);
}

void update(int x, int v)
{
    makeroot(x);
    val[x] = v;
    pushup(x);
}
```

使用说明：
- 路径信息：`split(x, y)` 后答案在 `sum[y]`（y 是 splay 根）；路径打标记同理挂在 y 上。
- `link / cut` 自带合法性检查，非法操作静默跳过；需要报错就自己改返回值。
- 下标从 1 开始，`fa` / `ch` 全局零初始化即可用。

变体：
- 路径最大值 / 覆盖赋值：加 `mx / laz` 懒标记，`pushdown` 里同步下传。
- 维护链上信息 + 子树信息（LCT 维护虚子树和）：`pushup` 需另开 `ssum` 记录虚儿子信息和，access 换重儿子时增减。
- 动态 MST / 动态直径：LCT 上维护边的最值（边权转点权：每条边建一个点）。
