# 02 · 数据结构

## 1. 树状数组：单点加 + 区间和

适用：下标从 1 开始，需要单点修改 + 区间求和（可推广二维）。

复杂度：修改 / 查询均 $O(\log n)$，空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int n;
ll c[MAXN];

int lowbit(int x)
{
    return x & (-x);
}

void add(int i, ll v)
{
    for (; i <= n; i += lowbit(i)) c[i] += v;
}

ll sum(int i)
{
    ll s = 0;
    for (; i > 0; i -= lowbit(i)) s += c[i];
    return s;
}

ll query(int l, int r)
{
    return sum(r) - sum(l - 1);
}
```

使用说明：
- 初始化：把原数组 a[] 的每个值 `add(i, a[i])`。
- 求逆序对：按值域离散化后从左到右 `query(a[i]+1, n)` 累加。
- 求第 k 小：配合 `lowbit` 倍增跳（见变体）。

变体：
- 区间加 + 单点查询：维护差分数组，`add(l,v), add(r+1,-v)`，单点查 `sum(i)`。
- 区间加 + 区间求和：开两个树状数组（$\text{差分}+\text{差分}\times i$）。
- 求全局第 k 小（权值树状数组）：倍增。

## 2. 线段树：区间加 + 区间求和

适用：任意区间修改（加 / 减）+ 区间查询。维护和的版本。

复杂度：$O(\log n)$ / 次操作，空间 $O(4n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
ll a[MAXN];
ll sum[MAXN << 2];
ll lazy[MAXN << 2];

void push_up(int rt)
{
    sum[rt] = sum[rt << 1] + sum[rt << 1 | 1];
}

void push_down(int rt, int l, int r)
{
    if (lazy[rt])
    {
        int mid = (l + r) >> 1;
        sum[rt << 1] += lazy[rt] * (mid - l + 1);
        sum[rt << 1 | 1] += lazy[rt] * (r - mid);
        lazy[rt << 1] += lazy[rt];
        lazy[rt << 1 | 1] += lazy[rt];
        lazy[rt] = 0;
    }
}

void build(int rt, int l, int r)
{
    lazy[rt] = 0;
    if (l == r)
    {
        sum[rt] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(rt << 1, l, mid);
    build(rt << 1 | 1, mid + 1, r);
    push_up(rt);
}

void update(int rt, int l, int r, int ql, int qr, ll v)
{
    if (ql <= l && r <= qr)
    {
        sum[rt] += v * (r - l + 1);
        lazy[rt] += v;
        return;
    }
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) update(rt << 1, l, mid, ql, qr, v);
    if (qr > mid) update(rt << 1 | 1, mid + 1, r, ql, qr, v);
    push_up(rt);
}

ll query(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return sum[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query(rt << 1, l, mid, ql, qr);
    if (qr > mid) res += query(rt << 1 | 1, mid + 1, r, ql, qr);
    return res;
}
```

使用说明：
- 调用：`build(1,1,n)`；区间加 `update(1,1,n,l,r,v)`；区间和 `query(1,1,n,l,r)`。
- 节点编号 1 开头，`rt<<1|1` 为右子。
- 修改为"赋值"时见模板 3。

变体：
- 维护区间最大值 / 最小值：`push_up` 改取 max/min，`sum` 换 `mx`，lazy 加法不变。
- 区间取反 / 异或：lazy 存 0/1，`sum = len - sum`。
- 动态开点 / 权值线段树：不用 build，`update` 时 `if(!rt) rt=++tot`。

## 3. 线段树：区间赋值 + 区间和 / 区间最值（覆盖型）

适用：操作是"区间整体赋成同一个值"（覆盖）而不是累加，如染色、区间清空重填。

复杂度：$O(\log n)$ / 次操作，空间 $O(4n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const ll NEG = 0x3f3f3f3f3f3f3f3fLL;
ll a[MAXN];
ll sum[MAXN << 2];
ll mx[MAXN << 2];
ll tag[MAXN << 2];

void apply(int rt, int l, int r, ll v)
{
    sum[rt] = v * (r - l + 1);
    mx[rt] = v;
    tag[rt] = v;
}

void push_down(int rt, int l, int r)
{
    if (tag[rt] != NEG)
    {
        int mid = (l + r) >> 1;
        apply(rt << 1, l, mid, tag[rt]);
        apply(rt << 1 | 1, mid + 1, r, tag[rt]);
        tag[rt] = NEG;
    }
}

void push_up(int rt)
{
    sum[rt] = sum[rt << 1] + sum[rt << 1 | 1];
    mx[rt] = max(mx[rt << 1], mx[rt << 1 | 1]);
}

void build(int rt, int l, int r)
{
    tag[rt] = NEG;
    if (l == r)
    {
        sum[rt] = mx[rt] = a[l];
        return;
    }
    int mid = (l + r) >> 1;
    build(rt << 1, l, mid);
    build(rt << 1 | 1, mid + 1, r);
    push_up(rt);
}

void assign(int rt, int l, int r, int ql, int qr, ll v)
{
    if (ql <= l && r <= qr)
    {
        apply(rt, l, r, v);
        return;
    }
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    if (ql <= mid) assign(rt << 1, l, mid, ql, qr, v);
    if (qr > mid) assign(rt << 1 | 1, mid + 1, r, ql, qr, v);
    push_up(rt);
}

ll query_sum(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return sum[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = 0;
    if (ql <= mid) res += query_sum(rt << 1, l, mid, ql, qr);
    if (qr > mid) res += query_sum(rt << 1 | 1, mid + 1, r, ql, qr);
    return res;
}

ll query_max(int rt, int l, int r, int ql, int qr)
{
    if (ql <= l && r <= qr) return mx[rt];
    push_down(rt, l, r);
    int mid = (l + r) >> 1;
    ll res = NEG;
    if (ql <= mid) res = max(res, query_max(rt << 1, l, mid, ql, qr));
    if (qr > mid) res = max(res, query_max(rt << 1 | 1, mid + 1, r, ql, qr));
    return res;
}
```

使用说明：
- 与模板 2 的区别：lazy 换成 `tag`，标记"整个区间被覆盖为 v"，`NEG` 表示无标记（注意别让真实值取到 NEG）。
- 先 build 再 assign / query。
- 若还要支持"区间加"和"区间赋值"同时存在，需两个懒标记并规定优先级（赋值覆盖加法）。

变体：
- 维护 gcd：`push_up` 取 gcd，区间加（差分）可做区间 gcd。
- 与线段树合并 / 扫描线结合做矩形面积并 / 周长。

## 4. 并查集：路径压缩 + 按秩合并

适用：动态连通性、最小生成树 Kruskal、离线处理区间关系。

复杂度：均摊 $O(\alpha(n))$（反阿克曼，近似常数），空间 $O(n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int fa[MAXN];
int rk[MAXN];

int find(int x)
{
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

void merge(int x, int y)
{
    x = find(x);
    y = find(y);
    if (x == y) return;
    if (rk[x] < rk[y]) swap(x, y);
    fa[y] = x;
    if (rk[x] == rk[y]) rk[x]++;
}

void init(int n)
{
    for (int i = 1; i <= n; i++)
    {
        fa[i] = i;
        rk[i] = 0;
    }
}
```

使用说明：
- 先 `init(n)`。`find` 带路径压缩（迭代版更快：`while(fa[x]!=x) x=fa[x]=fa[fa[x]]`）。
- 按秩合并保证树高 $O(\log n)$，配合路径压缩均摊常数。

变体：
- 带权并查集：维护到根的权值（食物链、种类），`find` 时累加权值。
- 可撤销并查集：用栈记录合并操作，按秩合并后支持回滚（离线分治）。

## 5. ST 表：静态区间最值 RMQ

适用：数组只读不改，大量区间 max/min 查询。不能用于求和（和没有幂等性）。

复杂度：预处理 $O(n\log n)$，查询 $O(1)$，空间 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int LOG = 18;
int st[MAXN][LOG];
int lg2[MAXN];

void init_st(int n)
{
    for (int i = 2; i <= n; i++) lg2[i] = lg2[i >> 1] + 1;
    for (int j = 1; (1 << j) <= n; j++) for (int i = 1; i + (1 << j) - 1 <= n; i++) st[i][j] = max(st[i][j - 1], st[i + (1 << (j - 1))][j - 1]);
}

int query_max(int l, int r)
{
    int k = lg2[r - l + 1];
    return max(st[l][k], st[r - (1 << k) + 1][k]);
}
```

使用说明：
- 先把原数组放入 `st[i][0]`，再调用 `init_st(n)`。
- LOG 取满足 $2^{\text{LOG}}\ge n$ 的最小值（n=1e5 用 17，留余量取 18）。

变体：
- 区间 gcd：`max` 换 `__gcd`，仍幂等可查。
- 区间 min：`max` 换 `min`。
- 树上 RMQ：欧拉序 + ST 表求 LCA（可代替倍增）。

## 6. 主席树：静态区间第 k 小

适用：静态数组，询问 [l,r] 内第 k 小（或 k 大、区间不同数个数等）。

复杂度：建树 $O(n\log n)$，单次查询 $O(\log n)$，空间 $O(n\log n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int ls[MAXN * 20];
int rs[MAXN * 20];
int cnt[MAXN * 20];
int root[MAXN];
int tot;

int build(int l, int r)
{
    int rt = ++tot;
    cnt[rt] = 0;
    if (l < r)
    {
        int mid = (l + r) >> 1;
        ls[rt] = build(l, mid);
        rs[rt] = build(mid + 1, r);
    }
    return rt;
}

int update(int pre, int l, int r, int pos)
{
    int rt = ++tot;
    ls[rt] = ls[pre];
    rs[rt] = rs[pre];
    cnt[rt] = cnt[pre] + 1;
    if (l < r)
    {
        int mid = (l + r) >> 1;
        if (pos <= mid) ls[rt] = update(ls[pre], l, mid, pos);
        else rs[rt] = update(rs[pre], mid + 1, r, pos);
    }
    return rt;
}

int query_kth(int u, int v, int l, int r, int k)
{
    if (l == r) return l;
    int mid = (l + r) >> 1;
    int left = cnt[ls[v]] - cnt[ls[u]];
    if (k <= left) return query_kth(ls[u], ls[v], l, mid, k);
    return query_kth(rs[u], rs[v], mid + 1, r, k - left);
}
```

使用说明：
- 值域离散化到 [1, m]，原数组 a[1..n]。
- `root[0] = build(1, m)`；对 i=1..n：`root[i] = update(root[i-1], 1, m, id(a[i]))`。
- 询问 [l,r] 第 k 小：`query_kth(root[l-1], root[r], 1, m, k)`，返回离散化下标，映射回原值。

变体：
- 求区间小于 x 的个数：`query_less` 统计前缀和小于某值的数量。
- 可持久化权值线段树可扩展到树上路径第 k 小（配合树剖 / dfs 序）。

## 7. 莫队：区间不同颜色数（离线）

适用：只有区间询问、无修改，维护的贡献可 $O(1)$ 增减（不同数、众数、出现次数奇偶等）。

复杂度：$O((n+q)\sqrt n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
int a[MAXN];
int cnt[MAXN];
ll ans[MAXN];
int ql[MAXN], qr[MAXN], qid[MAXN], blk[MAXN];
int ord[MAXN];
int n, m, q;
ll cur_ans;

bool cmp(int x, int y)
{
    if (blk[ql[x]] != blk[ql[y]]) return blk[ql[x]] < blk[ql[y]];
    return qr[x] < qr[y];
}

void add(int p)
{
    cnt[a[p]]++;
    if (cnt[a[p]] == 1) cur_ans++;
}

void del(int p)
{
    cnt[a[p]]--;
    if (cnt[a[p]] == 0) cur_ans--;
}

void solve()
{
    int size = max(1, (int)sqrt(n));
    for (int i = 1; i <= n; i++) blk[i] = (i - 1) / size + 1;
    for (int i = 1; i <= q; i++) ord[i] = i;
    sort(ord + 1, ord + q + 1, cmp);
    int L = 1, R = 0;
    cur_ans = 0;
    for (int i = 1; i <= q; i++)
    {
        int idx = ord[i];
        while (L > ql[idx])
        {
            add(--L);
        }
        while (R < qr[idx])
        {
            add(++R);
        }
        while (L < ql[idx])
        {
            del(L++);
        }
        while (R > qr[idx])
        {
            del(R--);
        }
        ans[qid[idx]] = cur_ans;
    }
}
```

使用说明：
- 询问存于 `ql/qr/qid`（下标 1..q），a[1..n] 为原数组。
- 调用 `solve()` 后答案写入 `ans[]`，按原询问顺序输出。
- 奇偶优化：`cmp` 里 R 的方向按块奇偶反转，常数减半（模板已可加）。

变体：
- 带修改莫队：增加时间维 $t$，块大小取 $n^{2/3}$，复杂度 $O(n^{5/3})$。
- 树上莫队：欧拉序转化为序列莫队。

## 8. 分块：区间加 + 区间和

适用：操作复杂（区间加、区间和、区间排序、区间众数等）线段树难以维护时，根号分治兜底。

复杂度：$O((n+q)\sqrt n)$。

```cpp
#include <bits/stdc++.h>
using namespace std;

#define ll long long

const int MAXN = 100005;
const int BS = 320;
ll a[MAXN];
ll sum[MAXN / BS + 5];
ll lazy[MAXN / BS + 5];
int L[MAXN / BS + 5], R[MAXN / BS + 5], bel[MAXN];
int n, q, bn;

void build()
{
    bn = (n + BS - 1) / BS;
    for (int i = 1; i <= bn; i++)
    {
        L[i] = (i - 1) * BS + 1;
        R[i] = min(i * BS, n);
        sum[i] = 0;
        for (int j = L[i]; j <= R[i]; j++)
        {
            bel[j] = i;
            sum[i] += a[j];
        }
    }
}

void update(int l, int r, ll v)
{
    int bl = bel[l], br = bel[r];
    if (bl == br)
    {
        for (int i = l; i <= r; i++)
        {
            a[i] += v;
            sum[bl] += v;
        }
        return;
    }
    for (int i = l; i <= R[bl]; i++)
    {
        a[i] += v;
        sum[bl] += v;
    }
    for (int i = bl + 1; i <= br - 1; i++) lazy[i] += v;
    for (int i = L[br]; i <= r; i++)
    {
        a[i] += v;
        sum[br] += v;
    }
}

ll query(int l, int r)
{
    int bl = bel[l], br = bel[r];
    ll res = 0;
    if (bl == br)
    {
        for (int i = l; i <= r; i++) res += a[i] + lazy[bl];
        return res;
    }
    for (int i = l; i <= R[bl]; i++) res += a[i] + lazy[bl];
    for (int i = bl + 1; i <= br - 1; i++) res += sum[i] + lazy[i] * (R[i] - L[i] + 1);
    for (int i = L[br]; i <= r; i++) res += a[i] + lazy[br];
    return res;
}
```

使用说明：
- 先 `build()` 再操作。`update(l,r,v)` 区间加，`query(l,r)` 区间和。
- 整块打 lazy，零散块暴力改并更新 sum。
- 块大小 BS 按 $n$ 调整，$\sqrt n$ 附近最优。

变体：
- 区间排序：块内维护有序 vector，查询二分，修改整块重建。
- 区间众数 / 出现次数：预处理块间答案 + 两侧暴力。
